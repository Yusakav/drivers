#### **1.2 CH32L103 I2C 底层驱动示例**

```c
/**
 * @file bq25710_platform.c
 * @brief CH32L103 平台特定的 BQ25710 I2C 驱动实现
 */

#include "ch32l103.h"
#include "bq25710.h"

/* I2C 外设定义 */
#define BQ25710_I2C_PORT    I2C1
#define BQ25710_I2C_CLK     RCC_APB1Periph_I2C1
#define BQ25710_I2C_SCL_PIN GPIO_Pin_6
#define BQ25710_I2C_SDA_PIN GPIO_Pin_7
#define BQ25710_I2C_GPIO_PORT GPIOB
#define BQ25710_I2C_GPIO_CLK RCC_APB2Periph_GPIOB

/* 超时定义 */
#define I2C_TIMEOUT_MS      100

/**
 * @brief 初始化 CH32L103 的 I2C1 外设
 */
static int platform_i2c_init(void)
{
    GPIO_InitTypeDef GPIO_InitStructure;
    I2C_InitTypeDef I2C_InitStructure;

    /* 使能时钟 */
    RCC_APB2PeriphClockCmd(BQ25710_I2C_GPIO_CLK, ENABLE);
    RCC_APB1PeriphClockCmd(BQ25710_I2C_CLK, ENABLE);

    /* 配置 GPIO 为复用开漏模式 */
    GPIO_InitStructure.GPIO_Pin = BQ25710_I2C_SCL_PIN | BQ25710_I2C_SDA_PIN;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_OD;  // 开漏输出
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(BQ25710_I2C_GPIO_PORT, &GPIO_InitStructure);

    /* I2C 配置 */
    I2C_InitStructure.I2C_ClockSpeed = 100000;     // 100kHz (标准模式)
    I2C_InitStructure.I2C_Mode = I2C_Mode_I2C;
    I2C_InitStructure.I2C_DutyCycle = I2C_DutyCycle_2;
    I2C_InitStructure.I2C_OwnAddress1 = 0x00;     // 从机地址(不使用)
    I2C_InitStructure.I2C_Ack = I2C_Ack_Enable;
    I2C_InitStructure.I2C_AcknowledgedAddress = I2C_AcknowledgedAddress_7bit;
    I2C_Init(BQ25710_I2C_PORT, &I2C_InitStructure);

    /* 使能 I2C */
    I2C_Cmd(BQ25710_I2C_PORT, ENABLE);

    return 0;  // 成功
}

/**
 * @brief I2C 读取操作 (阻塞模式)
 */
static int platform_i2c_read(uint8_t dev_addr, uint8_t reg_addr,
                              uint8_t *data, uint16_t len)
{
    uint32_t timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;

    /* 等待总线空闲 */
    while (I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_BUSY)) {
        if (--timeout == 0) return -1;  // 超时
    }

    /* 发送起始条件 */
    I2C_GenerateSTART(BQ25710_I2C_PORT, ENABLE);
    timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;
    while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_SB)) {
        if (--timeout == 0) return -1;
    }

    /* 发送设备地址 (写方向，用于设置寄存器地址) */
    I2C_SendData(BQ25710_I2C_PORT, (dev_addr << 1) & 0xFE);
    timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;
    while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_ADDR)) {
        if (--timeout == 0) return -2;  // 无应答
    }
    I2C_ClearFlag(BQ25710_I2C_PORT, I2C_FLAG_ADDR);

    /* 发送寄存器地址 */
    I2C_SendData(BQ25710_I2C_PORT, reg_addr);
    timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;
    while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_TXE)) {
        if (--timeout == 0) return -1;
    }
    while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_BTF)) {
        if (--timeout == 0) return -1;
    }

    /* 重复起始条件 (切换到读模式) */
    I2C_GenerateSTART(BQ25710_I2C_PORT, ENABLE);
    timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;
    while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_SB)) {
        if (--timeout == 0) return -1;
    }

    /* 发送设备地址 (读方向) */
    I2C_SendData(BQ25710_I2C_PORT, (dev_addr << 1) | 0x01);
    timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;
    while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_ADDR)) {
        if (--timeout == 0) return -2;
    }
    I2C_ClearFlag(BQ25710_I2C_PORT, I2C_FLAG_ADDR);

    /* 读取数据 */
    for (uint16_t i = 0; i < len; i++) {
        if (i == len - 1) {
            /* 最后一个字节，发送 NACK */
            I2C_AcknowledgeConfig(BQ25710_I2C_PORT, DISABLE);
            I2C_GenerateSTOP(BQ25710_I2C_PORT, ENABLE);
        } else {
            I2C_AcknowledgeConfig(BQ25710_I2C_PORT, ENABLE);
        }

        timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;
        while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_RXNE)) {
            if (--timeout == 0) return -1;
        }
        data[i] = I2C_ReceiveData(BQ25710_I2C_PORT);
    }

    /* 恢复 ACK (为下次传输准备) */
    I2C_AcknowledgeConfig(BQ25710_I2C_PORT, ENABLE);

    return 0;  // 成功
}

/**
 * @brief I2C 写入操作 (阻塞模式)
 */
static int platform_i2c_write(uint8_t dev_addr, uint8_t reg_addr,
                               const uint8_t *data, uint16_t len)
{
    uint32_t timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;

    /* 等待总线空闲 */
    while (I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_BUSY)) {
        if (--timeout == 0) return -1;
    }

    /* 发送起始条件 */
    I2C_GenerateSTART(BQ25710_I2C_PORT, ENABLE);
    timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;
    while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_SB)) {
        if (--timeout == 0) return -1;
    }

    /* 发送设备地址 (写方向) */
    I2C_SendData(BQ25710_I2C_PORT, (dev_addr << 1) & 0xFE);
    timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;
    while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_ADDR)) {
        if (--timeout == 0) return -2;  // 无应答
    }
    I2C_ClearFlag(BQ25710_I2C_PORT, I2C_FLAG_ADDR);

    /* 发送寄存器地址 */
    I2C_SendData(BQ25710_I2C_PORT, reg_addr);
    timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;
    while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_TXE)) {
        if (--timeout == 0) return -1;
    }

    /* 写入数据 */
    for (uint16_t i = 0; i < len; i++) {
        I2C_SendData(BQ25710_I2C_PORT, data[i]);
        timeout = I2C_TIMEOUT_MS * SystemCoreClock / 1000;
        while (!I2C_GetFlagStatus(BQ25710_I2C_PORT, I2C_FLAG_BTF)) {
            if (--timeout == 0) return -1;
        }
    }

    /* 发送停止条件 */
    I2C_GenerateSTOP(BQ25710_I2C_PORT, ENABLE);

    return 0;  // 成功
}

/**
 * @brief 注册 BQ25710 驱动到平台

    }
```

```c
void bq25710_temperature_compensation(int16_t temp_celsius)
{
    uint16_t target_current;
    uint16_t target_voltage;
    
    if (temp_celsius < 0) {
        /* 低温: 减半电流 */
        target_current = 1024;  // 1A
        target_voltage = 4100;  // 降低电压防止锂析出
    }
    else if (temp_celsius < 10) {
        /* 低温区: 限流 1.5A */
        target_current = 1536;
        target_voltage = 4150;
    }
    else if (temp_celsius > 45) {
        /* 高温: 降额充电 */
        target_current = 2048;  // 2A
        target_voltage = 4150;  // 降低浮充电压延长寿命
    }
    else if (temp_celsius > 55) {
        /* 超高温: 停止充电 */
        bq25710_disable_charging();
        return;
    }
    else {
        /* 正常温度范围 (10-45°C): 全速充电 */
        target_current = 4096;  // 4A
        target_voltage = 4200;  // 标准电压
    }
    
    bq25710_set_charge_current(target_current);
    bq25710_set_max_charge_voltage(target_voltage);
}
```

#### **3.2 故障恢复策略**

```c
/**
 * @brief 智能故障处理与恢复函数
 * @return 处理结果: 0-已恢复, 1-需人工干预, -1-致命错误
 */
int bq25710_fault_recovery_handler(void)
{
    uint16_t fault_mask;
    BQ25710_ChargerStatus_t status;
    
    bq25710_get_charger_status(&status);
    fault_mask = status.all & 0xFF;
    
    if (fault_mask == 0) {
        return 0;  // 无故障
    }
    
    /* 分析故障类型并尝试自动恢复 */
    
    if (status.bits.FAULT_ACOV) {
        /* 输入过压: 可能是适配器损坏或静电 */
        bq25710_log("[FAULT] Input Overvoltage detected!");
        bq25710_clear_faults();
        
        /* 等待 2 秒后重新检测 */
        delay_ms(2000);
        if (bq25710_is_adapter_present() > 0) {
            bq25710_log("[RECOVERY] Adapter reconnected safely");
            return 0;
        }
        return 1;  // 需要更换适配器
    }
    
    if (status.bits.FAULT_ACOC) {
        /* 输入过流: 降低输入电流限制 */
        uint16_t current_lim;
        bq25710_get_input_current_limit(&current_lim);
        current_lim = (current_lim * 8) / 10;  // 降到 80%
        current_lim = (current_lim / 50) * 50;  // 对齐步进
        
        bq25710_log("[FAULT] Input Overcurrent! Reducing limit to %dmA", current_lim);
        bq25710_set_input_current_limit(current_lim);
        bq25710_clear_faults();
        return 0;
    }
    
    if (status.bits.FAULT_BATOC) {
        /* 电池放电过流: 可能有短路 */
        bq25710_log("[FAULT] Battery discharge overcurrent!");
        bq25710_disable_charging();  // 先停止充电
        bq25710_enter_hiz_mode(0);   // 进入高阻态隔离电池
        bq25710_clear_faults();
        return 1;  // 需要检查负载
    }
    
    if (status.bits.FAULT_SYS_SHORT) {
        /* 系统严重短路: 7次重启都失败 */
        bq25710_log("[FATAL] System short circuit detected!");
        bq25710_disable_charging();
        bq25710_enter_hiz_mode(1);  // 强制放电关机
        return -1;  // 致命错误，需硬件检修
    }
    
    if (status.bits.FAULT_OTG_OVP || status.bits.FAULT_OTG_UVP) {
        /* OTG 输出异常 */
        bq25710_log("[FAULT] OTG voltage abnormal");
        bq25710_disable_otg();
        bq25710_clear_faults();
        return 0;
    }
    
    /* 其他未知故障: 清除并记录 */
    bq25710_log("[FAULT] Unknown fault mask: 0x%02X", fault_mask);
    bq25710_clear_faults();
    return 0;
}
```

### 📊 **四、ADC 数据校准与精度优化**

#### **4.1 系统校准流程**

```c
typedef struct {
    int16_t vbat_offset_mV;    // VBAT 零偏 (mV)
    float  vbat_gain;          // VBAT 增益系数
    int16_t ichg_offset_mA;    // ICHG 零偏 (mA)
    float  ichg_gain;          // ICHG 增益系数
} BQ25710_Calibration_t;

static BQ25710_Calibration_t s_calibration = {0};

/**
 * @brief 执行两点校准 (需要已知精度的参考源)
 * @param point1_vbat: 校准点1的已知电压 (mV)
 * @param point1_adc:  校准点1的ADC原始值
 * @param point2_vbat: 校准点2的已知电压 (mV)
 * @param point2_adc:  校准点2的ADC原始值
 */
void bq25710_calibrate_vbat(uint16_t point1_vbat, uint16_t point1_adc,
                            uint16_t point2_vbat, uint16_t point2_adc)
{
    /* 计算增益: gain = Δ真实值 / ΔADC值 */
    s_calibration.vbat_gain = (float)(point2_vbat - point1_vbat) / 
                              (float)(point2_adc - point1_adc);
    
    /* 计算偏移: offset = 真实值 - ADC×gain */
    s_calibration.vbat_offset_mV = point1_vbat - (int16_t)(point1_adc * s_calibration.vbat_gain);
}

/**
 * @brief 获取校准后的精确电池电压
 */
uint16_t bq25710_get_calibrated_vbat(void)
{
    uint16_t raw_adc;
    bq25710_adc_read_channel(BQ25710_ADC_VBAT, &raw_adc);
    
    /* 应用校准公式: V_calibrated = ADC × Gain + Offset */
    int16_t calibrated = (int16_t)(raw_adc * s_calibration.vbat_gain) + 
                          s_calibration.vbat_offset_mV;
    
    return (uint16_t)(calibrated > 0 ? calibrated : 0);
}
```

#### **4.2 数字滤波算法 (滑动平均)**

```c
#define ADC_FILTER_DEPTH  8

typedef struct {
    uint16_t buffer[ADC_FILTER_DEPTH];
    uint8_t  index;
    uint8_t  filled;
} BQ25710_ADC_Filter_t;

static BQ25710_ADC_Filter_t s_vbat_filter = {0};

/**
 * @brief 滑动平均滤波器 (消除噪声尖峰)
 */
uint16_t adc_filter_update(BQ25710_ADC_Filter_t *filter, uint16_t new_value)
{
    filter->buffer[filter->index] = new_value;
    filter->index = (filter->index + 1) % ADC_FILTER_DEPTH;
    
    if (filter->filled < ADC_FILTER_DEPTH) {
        filter->filled++;
    }
    
    /* 计算平均值 */
    uint32_t sum = 0;
    uint8_t count = (filter->filled < ADC_FILTER_DEPTH) ? filter->filled : ADC_FILTER_DEPTH;
    
    for (uint8_t i = 0; i < count; i++) {
        sum += filter->buffer[i];
    }
    
    return (uint16_t)(sum / count);
}

/* 使用示例: 在主循环中定期调用 */
void monitoring_task(void)
{
    static uint32_t last_tick = 0;
    
    if (get_tick() - last_tick >= 100) {  // 每 100ms 采样一次
        last_tick = get_tick();
        
        uint16_t raw_vbat;
        bq25710_adc_read_channel(BQ25710_ADC_VBAT, &raw_vbat);
        
        uint16_t filtered_vbat = adc_filter_update(&s_vbat_filter, raw_vbat);
        
        printf("VBAT(raw)=%d mV, VBAT(filtered)=%d mV\r\n", raw_vbat, filtered_vbat);
    }
}
```

```c
/**
 * @brief EXTI0/1 中断服务程序 (CHRG_OK + PROCHOT)
 */

{

        /* CHRG_OK 中断: 适配器插拔事件 */
        g_chrg_ok_event = 1;
        EXTI_ClearITPendingBit(EXTI_Line0);
    }

    if (EXTI_GetITStatus(EXTI_Line1) != RESET) {
        /* PROCHOT 中断: 过功率警告 */
        g_prochot_event = 1;
        EXTI_ClearITPendingBit(EXTI_Line1);
    }
}

 * @brief 事件处理主循环 (在 main() 的 while(1) 中调用)

    if (g_chrg_ok_event) {
        g_chrg_ok_event = 0;
        
        int adapter = bq25710_is_adapter_present();
        if (adapter > 0) {
            printf("[EVENT] Adapter connected!\r\n");
            
            /* 可选: 重新应用 ICO 或更新充电参数 */
            bq25710_enable_ico(1);
        } else {
            printf("[EVENT] Adapter removed!\r\n");
            
            /* 适配器拔出时的清理工作 */
            bq25710_enable_ico(0);
            
            /* 检查是否需要进入低功耗 */
            BQ25710_ADC_Result_t adc;
            if (bq25710_adc_read_all(&adc) == BQ25710_OK) {
                if (adc.vbat_mV < 3700) {  // 低电量警告
                    printf("[WARN] Low battery: %dmV\r\n", adc.vbat_mV);
                    /* 可以在这里触发保存数据等操作 */
                }
            }
        }
    }
    
    /* 处理 PROCHOT 报警事件 */
    if (g_prochot_event) {
        g_prochot_event = 0;
        
        BQ25710_ProchotStatus_t prochot;
        bq25710_get_prochot_status(&prochot);
        
        printf("[ALERT] PROCHOT triggered! Status=0x%04X\r\n", prochot.all);
        
        /* 分析具体原因 */
        if (prochot.bits.STAT_ICRIT) {
            printf("  -> Input peak overcurrent!\r\n");
            /* 自动降低输入电流 */
            uint16_t iin;
            bq25710_get_input_current_limit(&iin);
            bq25710_set_input_current_limit(iin * 80 / 100);  // 降额20%
        }
        if (prochot.bits.STAT_VDPM) {
            printf("  -> VBUS undervoltage!\r\n");
            /* 可能是适配器过载 */
        }
        
        /* 清除 PROCHOT 脉冲 */
        prochot.bits.PROCHOT_CLEAR = 0;
        bq25710_write_reg16(BQ25710_REG_PROCHOT_STATUS, prochot.all);
    }
```

### 📦 **六、完整的单电池充电管理器示例**

```c
/**
 * @file battery_manager.c
 * @brief 基于 BQ25710 的智能电池管理系统
 *
 * 功能特性:
 * - 自动识别电池串数和健康状态
 * - 四阶段充电控制 (TRICKLE/CC/CV/TOP-OFF)
 * - 温度监控与补偿
 * - 充电统计 (容量估算、循环计数)
 * - 老化均衡策略
 */

#include "bq25710.h"
#include <stdio.h>

/* ==================== 数据结构 ==================== */

typedef enum {
    CHG_STAGE_IDLE,           // 空闲
    CHG_STAGE_TRICKLE,        // 预充电 (涓流)
    CHG_STAGE_CC,             // 恒流充电 (快充)
    CHG_STAGE_CV,             // 恒压充电
    CHG_STAGE_TOP_OFF,        // 补偿充电 (截止)
    CHG_STAGE_COMPLETE,       // 充满
    CHG_STAGE_FAULT           // 故障暂停
} ChargeStage_t;

typedef struct {
    ChargeStage_t stage;
    uint32_t      stage_start_tick;
    uint32_t      total_charge_time_s;
    uint32_t      full_charge_count;
    uint16_t      design_capacity_mAh;   // 设计容量
    uint16_t      remaining_capacity_mAh;// 剩余容量估算
    uint16_t      cycle_count;           // 充放电循环次数
    int16_t       temperature_c;         // 电池温度
} BatteryManager_t;

static BatteryManager_t s_batt_mgr = {0};

/* ==================== 状态机实现 ==================== */

/**
 * @brief 充电状态机核心逻辑 (每秒调用一次)
 */
void battery_manager_task_1s(void)
{
    BQ25710_ADC_Result_t adc;
    BQ25710_ChargerStatus_t status;
    uint16_t vbat, ichg;
    
    /* 读取所有 ADC 通道 */
    if (bq25710_adc_read_all(&adc) != BQ25710_OK) {
        return;  // 通信错误，跳过本次
    }
    
    vbat = adc.vbat_mV;
    ichg = adc.ichg_mA;
    
    /* 获取充电器状态 */
    bq25710_get_charger_status(&status);
    
    /* 检查是否有适配器 */
    if (!status.bits.AC_STAT) {
        /* 无适配器: 进入放电模式 */
        s_batt_mgr.stage = CHG_STAGE_IDLE;
        update_discharge_estimation(&adc);
        return;
    }
    
    /* ===== 状态机转换逻辑 ===== */
    
    switch (s_batt_mgr.stage) {
        case CHG_STAGE_IDLE:
            if (vbat < 3000) {
                /* 深度欠压: 进入预充电 */
                s_batt_mgr.stage = CHG_STAGE_TRICKLE;
                bq25710_set_charge_current(128);  // 128mA 涓流
                bq25710_set_max_charge_voltage(3000);  // 限制在 3V
                s_batt_mgr.stage_start_tick = get_tick();
            }
            else if (vbat < 4200) {
                /* 正常电压: 直接进入 CC 阶段 */
                s_batt_mgr.stage = CHG_STAGE_CC;
                apply_cc_parameters(vbat);
                s_batt_mgr.stage_start_tick = get_tick();
            }
            break;
            
        case CHG_STAGE_TRICKLE:
            /* 预充电阶段: 等待电压上升到 3.0V */
            if (vbat >= 3000) {
                s_batt_mgr.stage = CHG_STAGE_CC;
                apply_cc_parameters(vbat);
                s_batt_mgr.stage_start_tick = get_tick();
            }
            else if (ichg < 64) {
                /* 电流几乎为零但电压不升: 电池可能损坏 */
                s_batt_mgr.stage = CHG_STAGE_FAULT;
                printf("[ERROR] Precharge failed: dead battery?\r\n");
            }
            break;
            
        case CHG_STAGE_CC:
            /* 恒流阶段: 监测电压是否达到设定值 */
            if (vbat >= 4180) {  // 接近目标电压 (98%)
                s_batt_mgr.stage = CHG_STAGE_CV;
                s_batt_mgr.stage_start_tick = get_tick();
                
                /* 切换到 CV 参数 */
                bq25710_set_max_charge_voltage(4200);
                bq25710_set_charge_current(2048);  // 降低电流减少发热
            }
            break;
            
        case CHG_STAGE_CV:
            /* 恒压阶段: 等待电流降低到终止阈值 */
            if (ichg < 128) {  // 终止电流: C/20 或 128mA 取大者
                s_batt_mgr.stage = CHG_STAGE_TOP_OFF;
                s_batt_mgr.stage_start_tick = get_tick();
                
                /* 进入补电阶段: 维持 30 分钟确保完全充满 */
                bq25710_set_charge_current(256);
            }
            break;
            
        case CHG_STAGE_TOP_OFF:
            /* 补偿充电: 固定时间后结束 */
            if (get_tick_diff(s_batt_mgr.stage_start_tick) >= 30*60*1000) {  // 30分钟
                s_batt_mgr.stage = CHG_STAGE_COMPLETE;
                on_charge_complete();
            }
            break;
            
        case CHG_STAGE_COMPLETE:
            /* 已充满: 维持浮充或停止 */
            if (vbat < 4100) {
                /* 电压跌落超过 100mV: 重新启动充电 */
                s_batt_mgr.stage = CHG_STAGE_CC;
                apply_cc_parameters(vbat);
            }
            else {
                /* 保持微电流补充自放电损失 */
                bq25710_set_charge_current(64);
            }
            break;
            
        case CHG_STAGE_FAULT:
            /* 故障状态: 等待人工干预或超时重试 */
            if (get_tick_diff(s_batt_mgr.stage_start_tick) >= 60*60*1000) {  // 1小时
                /* 尝试自动恢复 */
                bq25710_clear_faults();
                s_batt_mgr.stage = CHG_STAGE_IDLE;
            }
            break;
    }
    
    /* 更新统计数据 */
    if (s_batt_mgr.stage >= CHG_STAGE_TRICKLE && 
        s_batt_mgr.stage <= CHG_STAGE_COMPLETE) {
        s_batt_mgr.total_charge_time_s++;
        
        /* 库仑计法估算充电量 */
        s_batt_mgr.remaining_capacity_mAh += (ichg / 3600);  // 每秒累加
        if (s_batt_mgr.remaining_capacity_mAh > s_batt_mgr.design_capacity_mAh) {
            s_batt_mgr.remaining_capacity_mAh = s_batt_mgr.design_capacity_mAh;
        }
    }
}

/**
 * @brief 根据电池电压动态调整 CC 阶段参数
 */
static void apply_cc_parameters(uint16_t vbat_mV)
{
    uint16_t charge_current;
    
    if (vbat_mV < 3500) {
        charge_current = 4096;  // 低电压时全速充电
    }
    else if (vbat_mV < 4000) {
        charge_current = 3072;  // 中间段保持较高电流
    }
    else {
        charge_current = 2048;  // 接近满电时降低电流
    }
    
    bq25710_set_charge_current(charge_current);
    bq25710_set_max_charge_voltage(4200);
}

/**
 * @brief 充电完成回调
 */
static void on_charge_complete(void)
{
    s_batt_mgr.full_charge_count++;
    s_batt_mgr.cycle_count++;
    s_batt_mgr.remaining_capacity_mAh = s_batt_mgr.design_capacity_mAh;
    
    printf("\n*** CHARGE COMPLETE ***\r\n");
    printf("Total time: %lu min\r\n", s_batt_mgr.total_charge_time_s / 60);
    printf("Full cycles: %u\r\n", s_batt_mgr.full_charge_count);
    printf("Battery health: %d%%\r\n", 
           100 * s_batt_mgr.design_capacity_mAh / (s_batt_mgr.cycle_count > 0 ? 
               s_batt_mgr.design_capacity_mAh : s_batt_mgr.design_capacity_mAh));
}

/**
 * @brief 放电期间更新剩余容量估算
 */
static void update_discharge_estimation(const BQ25710_ADC_Result_t *adc)
{
    /* 简化的库仑计: 根据放电电流递减 */
    if (adc->idchg_mA > 0) {
        s_batt_mgr.remaining_capacity_mAh -= (adc->idchg_mA / 3600);
        if ((int16_t)s_batt_mgr.remaining_capacity_mAh < 0) {
```

### 📚 **一、基础概念篇**

#### **1. VBUS (Voltage Bus)**
```
中文: 总线电压 / 母线电压
定义: USB 或适配器输出的直流电压线路
类比: 就像家里的"电源插座"，电能从这里流入设备
实际值: 通常 5V(USB)、9V/12V/20V(快充)
位置: BQ25710 的 VBUS 引脚连接到这里
```

#### **2. VSYS (System Voltage)**
```
中文: 系统电压
定义: 给 MCU/CPU 等系统负载供电的电压总线
类比: 就像小区的"配电箱"，把电分配给各个住户（屏幕、WiFi、CPU）
特点: 即使电池没电或拔掉，VSYS 也要保持供电（不能断电）
范围: 一般维持在 3.5V ~ 4.4V 左右
```

#### **3. VBAT (Battery Voltage)**
```
中文: 电池电压
定义: 锂电池正负极两端的实际电压
类比: 就像电池的"剩余体力值"
典型值:
  - 4.20V = 100% 满电（就像人吃饱了）
  - 3.70V = 50% 半电（刚吃完午饭）
  - 3.00V = 0%   没电了（饿得头晕）
  - 2.50V = 过放损坏（进医院抢救）
```

### ⚡ **二、电流相关术语**

#### **4. IIN (Input Current) - 输入电流**
```
中文: 从适配器流进的电流
公式: IIN = 从 VBUS 流入芯片的电流大小
单位: mA (毫安) 或 A (安培)
类比: 就像水管的"进水量"
      - 你用 5V/2A 的充电器 → IIN 最大 2000mA
      - 芯片自己决定实际取多少电流
```

#### **5. ICHG (Charge Current) - 充电电流**
```
中文: 流入电池进行充电的电流
方向: VSYS → 电池（给电池"喂饭"）
设置范围: 64mA ~ 8128mA (步进 64mA)
类比: 就像给电池"吃饭的速度"
      - 吃太快（电流大）→ 发热大，但充得快
      - 吃太慢（电流小）→ 不发热，但要等很久
      - 推荐值: 0.5C ~ 1C（C=电池容量）
        例: 3000mAh 电池 → 推荐 1500~3000mA
```

#### **6. IDCHG (Discharge Current) - 放电电流**
```
中文: 从电池流出的电流（给系统供电时）
方向: 电池 → VSYS（电池在"干活输出能量")
场景: 
  - 拔掉充电器后，电池通过 IDCHG 给手机供电
  - OTG 模式下，电池放电变成充电宝
测量: 通过 IBAT 引脚检测（需要外部采样电阻）
```

#### **7. IINDPM (Input Current DPM) - 输入电流动态电源管理**
```
全称: Input Current Dynamic Power Management
中文名: 输入电流动态调节/限流功能

【核心作用】防止把充电器"拉死"！
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ 问题场景:
  你的充电器只能提供 5V/1A (5瓦功率)
  但你设置了 IIN_HOST = 2000mA (2A)
  
  结果: 
  5V × 2A = 10W > 5W (充电器过载)
  → 充电器保护关断 → 手机突然没电！

✅ IINDPM 的智能保护:
  芯片自动检测到充电器撑不住了
  → 自动把电流降到 800mA (5V×0.8A=4W < 5W)
  → 充电器安全工作，虽然慢点但不会断
  
【两个关键寄存器】:
┌─────────────────────────────────────┐
│ IIN_HOST (寄存器 0x3F)              │
│ └─ 你设置的"期望最大值"             │
│    "我希望最多取这么多电"           │
│    例如: 设置 2000mA                │
├─────────────────────────────────────┤
│ IIN_DPM  (寄存器 0x22, 只读)       │
│ └─ 芯片实际生效的"真实限制值"       │
│    "充电器实际上只能给我这么多"     │
│    实际可能只有 800mA               │
│    (IIN_DPM ≤ IIN_HOST, 永远成立)  │
└─────────────────────────────────────┘
```

#### **8. ICO (Input Current Optimization) - 输入电流优化**
```
全称: Input Current Optimization
中文名: 输入电流自动优化算法

【解决的问题】:
  不同充电器能力不同，怎么知道它最大能供多少电流？
  
  - 原装充电器: 5V/2A → 可以取 2000mA
  - 电脑USB口:  5V/0.5A → 只能取 500mA
  - 劣质充电器: 5V/1A 但虚标 → 可能 800A就保护了

【ICO 算法的工作流程】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
步骤1: 用户调用 bq25710_enable_ico(1)

步骤2: 芯片开始"试探"充电器
       ┌──────────────────────────┐
       │ 取 500mA → 充电器OK ✓    │
       │ 取 1000mA → 充电器OK ✓   │
       │ 取 1500mA → 充电器OK ✓   │
       │ 取 2000mA → 充电器电压跌落 ✗ ← 发现极限！
       └──────────────────────────┘

步骤3: 自动回退到最后成功的值 (1500mA)

步骤4: 将这个最优值写入 IIN_DPM 寄存器

步骤5: ICO_DONE 标志置位，通知 MCU "我测好了"

【使用场景】:
  - 不知道用户会插什么充电器时（通用性强）
  - 支持 PD/QC 快充协议协商后重新测量
  - 首次上电初始化时自动执行一次
```

#### **9. VINDPM (Voltage Input DPM) - 输入欠压保护阈值**
```
全称: Voltage Input Dynamic Power Management
中文名: 输入电压动态调节门限/输入低压保护点

【为什么需要它？】
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
想象一个场景:

  你有一个 5V/1A 的充电器（内阻较大）
  当你的手机以 1.5A 大电流充电时:
  
  充电器内部压降 = 1.5A × 内阻 ≈ 0.5V
  实际到达芯片的电压 = 5V - 0.5V = 4.5V
  
  如果继续增大电流:
  4.5V → 4.0V → 3.5V → 2.8V (芯片无法工作了！)
  
  结果: 系统崩溃重启，反复循环 😵

【VINDPM 的保护机制】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
设置一个"最低电压警戒线"，例如 4.6V:

  正常状态: VBUS = 5.0V > 4.6V ✓ → 全速充电
  
  异常情况: VBUS 跌落到 4.55V < 4.6V ⚠️
           ↓
  芯片检测到触发了 VINDPM 保护
           ↓
  自动降低充电电流 (比如从 1500mA 降到 800mA)
           ↓
  VBUS 回升到 4.75V > 4.6V ✓ → 恢复正常
           ↓
  在这个新平衡点稳定工作（虽然慢但不会断）

【参数设置】:
  范围: 3840mV ~ 23552mV (步进 64mV)
  
  常用推荐值:
  ┌──────────────┬─────────────┬────────────────┐
  │ 输入源类型    │ VINDPM 设定值 │ 说明            │
  ├──────────────┼─────────────┼────────────────┤

  │ QC 9V        │ 8320 mV     │ 9V-0.68V 余量   │
  │ PD 12V       │ 11008 mV    │ 12V-1V 余量     │
  │ PD 20V       │ 18432 mV    │ 20V-1.6V 余量   │
  │ 弱适配器     │ 高一些       │ 更早触发保护    │
  └──────────────┴─────────────┴────────────────┘
```

### 🛡️ **四、保护机制术语**

#### **11. ACOC (Adapter Current Over Current) - 适配器过流保护**
```
全称: Adapter Current Over Current
中文名: 输入过流保护

【触发条件】: 输入电流超过阈值的 133% 或 200%
【硬件响应时间】: < 1 微秒（极快！）

【两种阈值可选】:
  Register 0x31 [ACOC_VTH]:
  ├── 0: 触发点 = ILIM2 × 133%  (较敏感，更早保护)
  └── 1: 触发点 = ILIM2 × 200%  (较宽松，允许峰值)

【举例说明】:
  你设置了 IIN_HOST = 2000mA
  ILIM2 默认为 IIN_HOST 的 150% = 3000mA
  
  如果 ACOC_VTH = 0 (133%模式):
  触发电流 = 3000mA × 1.33 = 3990mA (~4A)
  → 当瞬间电流冲到 4A 时立即切断
  
  如果 ACOC_VTH = 1 (200%模式):
  触发电流 = 3000mA × 2.0 = 6000mA (6A)
  → 允许更大的瞬态冲击（适合电机启动等场景）
```

#### **12. ACOV (Adapter Voltage Over Voltage) - 适配器过压保护**
```
全称: Adapter Voltage Over Voltage
中文名: 输入过压保护

【保护对象】: 防止高压烧毁芯片和后级电路
【默认阈值】: VBUS > 28V (不同型号略有差异)

【常见触发原因】:
  ❌ 使用了错误的充电器 (12V车充插到5V设备)
  ❌ 静电击穿 (ESD)
  ❌ 雷击浪涌
  ❌ 充电器内部故障导致输出失控

【保护动作】:
  1. 立即断开输入开关管 (Q1/Q4 关断)
  2. 设置 FAULT_ACOV 故障标志
  3. 等待电压恢复正常或手动清除故障
```

#### **13. BATOC (Battery Over Current) - 电池过流保护**
```
全称: Battery Over Current
中文名: 电池放电过流保护

【监测方向】: 电池 → 系统 (IDCHG 方向)
【应用场景】: 
  - OTG 模式下外接短路负载
  - 系统内部短路故障
  - 电池直接被意外短接

【两种阈值】:
  Register 0x31 [BATOC_VTH]:
  ├── 0: 触发点 = IDCHG设定值的 133%
  └── 1: 触发点 = IDCHG设定值的 200%

【与 ACOC 的区别】:
  ┌──────────┬─────────────┬──────────────┐
  │          │ ACOC        │ BATOC        │
  ├──────────┼─────────────┼──────────────┤
  │ 方向      │ 适配器→芯片  │ 电池→芯片    │
  │ 保护对象  │ 防止烧适配器 │ 防止损坏电池 │
  │ 触发条件  │ IIN过大     │ IDCHG过大    │
  │ 后果      │ 切断输入    │ 切断电池FET  │
  └──────────┴─────────────┴──────────────┘
```

#### **14. PROCHOT (Processor Hot) - 处理器过热报警**
```
全文名称: Processor Hot (or Power Hot)
中文名: 过功率/过热报警信号

【物理引脚】: 一个开漏输出引脚 (低电平有效)

【核心作用】: 告诉 CPU "我要撑不住了！请降频！"

【完整报警链路】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  [异常事件发生]
       ↓
  BQ25710 检测到以下任一条件:
  ├── ICRIT: 输入电流超过临界值 (如 >5A)
  ├── INOM:  输入平均电流超限 (持续高负载)
  ├── VDPM:  VBUS 电压过低 (适配器过载)
  ├── VSYS:  系统电压下冲 (瞬时大功耗)
  ├── IDCHG: 电池放电超流
  └── COMP:  外部比较器触发 (用户自定义条件)
       ↓
  PROCHOT 引脚输出低电平脉冲 (或持续拉低)
       ↓
  CPU 收到中断:
  ├── 降低主频 (从 2GHz 降到 800MHz)
  ├── 关闭部分核心 (关闭大核，只用小核)
  ├── 降低屏幕亮度
  └── 通知应用层进入省电模式
       ↓
  系统功耗下降 → 电流降低 → 危机解除
       ↓
  PROCHOT 释放 (恢复高电平)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

【脉冲宽度可配置】:
  Register 0x34 [PROCHOT_WIDTH]:
  ├── 00b: 100μs  (极短提醒)
  ├── 01b: 1ms    (标准脉冲)
  ├── 10b: 10ms   (明显警告)
  └── 11b: 5ms    (中等时长)

【扩展模式 (EN_PROCHOT_EXIT)]:
  如果置位此标志, PROCHOT 会一直拉低
  直到软件主动写寄存器清除
  (适用于严重故障需要人工介入的场景)
```

#### **15. ILIM2 / ICRIT / INOM - 三级电流监控**
```
这是三个递进的电流阈值，形成"三级防护":

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  电流值从小到大排列:

  Level 1: INOM (Normal Operating Margin)
  ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  含义: 正常工作余量边界
  阈值: 比 IIN_HOST 高 10% (固定)
  触发后果: 可选是否触发 PROCHOT (PP_INOM 位控制)
  
  类比: "水位到了警戒线，注意观察"
  

  Level 2: ICRIT (Critical Current)
  ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  含义: 临界电流 (即将危险)
  阈值: 由寄存器 0x33[15-11] 设置 (110%~450% of ILIM)
  默认值: ILIM2 × 150%
  触发后果: 必须触发 PROCHOT (PP_ICRIT 位默认使能)
  抗尖峰时间: 可配置 15us ~ 1ms (ICRIT_DEG)
  
  类比: "洪水要漫堤了，立刻报警！"


  Level 3: ILIM2 (Limit Level 2 - 硬件极限)
  ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  含义: 绝对不允许超过的物理极限
  阈值: 寄存器 0x33[15-15] (5位, 110%~450%)
  触发后果: 
    - ACOC/BATOC 硬件比较器动作
    - 强制关断开关管 (微秒级响应)
    - 设置不可屏蔽的故障锁存
  
  类比: "堤坝溃决，紧急闸门落下！"


  【三者关系图示】:
  
  电流(A)
    ^
  5 │                         ★ ILIM2 (绝对极限)
    │                        ╱
  4 │                       ╱  ← ACOC/BATOC 触发
    │                      ╱
  3 │   ──────────────────╱★★★ ICRIT (临界警告)
    │   ╱                 ╱     ← PROCHOT 必触发
  2 │  ╱  ★★★★★★★★★★★★╱       
    │ ╱   INOM (正常余量)          ← PROCHOT 可选
  1 │╱_____________________________ IIN_HOST (设定值)
    └──────────────────────────→ 时间
    
    安全区  │  警告区  │  危险区  │  禁区
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
```

#### **16. Power Path (电源路径管理)**
```
中文名: 电源通路管理

【传统充电架构 vs Power Path 架构】

❌ 传统架构 (无 Power Path):
  ┌──────┐     ┌──────┐
  │适配器 │────→│ 电池 │────→ 系统(MCU/屏幕...)
  └──────┘     └──────┘
  
  问题:
  - 电池坏了 → 整个系统无法工作
  - 电池没电时插入适配器 → 要等一会儿才能开机
  - 电池老化内阻增大 → 系统电压不稳

✅ BQ25710 的 Power Path 架构:
  ┌──────┐          
  │适配器 │────┐     
  └──────┘    │     
               ▼     
          ┌────────┐
          │ BQ25710 │ ← 智能调度中心
          └───┬┬┬───┘
              │ │ │
     ┌────────┘ │ └────────┐
     ▼          ▼          ▼
  ┌──────┐   ┌──────┐   ┌──────┐
  │ 系统 │   │ 电池 │   │ OTG  │
  └──────┘   └──────┘   └──────┘

  优势:
  ✅ 即使电池损坏/拆除，系统也能靠适配器独立运行
  ✅ 插上适配器瞬间就能开机 (不用等电池充到一定电压)
  ✅ 电池老化不影响系统稳定性 (VSYS由芯片稳压)
  ✅ 可以同时给系统供电 + 给电池充电 (分流)
```

#### **17. LDO Mode (Low Dropout Mode) - 低压差模式**
```
中文名: 低压差线性稳压模式 / 预充维持模式

【什么时候需要它？】
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
场景: 电池深度放电 (VBAT < 3.0V)

  正常情况下:
  适配器(5V) → BQ25710降压→ 系统(3.8V) → 电池(涓流充电)
  
  但如果电池电压极低 (例如 2.5V):
  开关转换器可能无法正常启动 (需要最小输入输出压差)
  
  此时 LDO 模式的作用:
  用线性方式 (不是开关方式) 维持一个最低系统电压
  让 MCU 能启动并开始预充电程序

【工作原理简化】:
  ┌─────────────────────────────────────┐
  │  正常模式 (Switching, 高效):        │
  │  VIN=5V → PWM调制 → VSYS=3.8V       │
  │  效率: 90%+                         │
  │                                     │
  │  LDO模式 (Linear, 低效但可靠):      │
  │  VIN=5V → 三极管线性降压→ VSYS=3.0V │
  │  效率: 60% (发热大，但能工作)        │
  └─────────────────────────────────────┘

【配置方法】:
  Register 0x12 [EN_LDO] = 1  (默认就是使能的)
  
  适用时机: 仅在电池电压 < 3.0V 时激活
  一旦电池电压回升到 3.0V 以上，自动切回高效开关模式
```

#### **18. Min System Voltage (最小系统电压)**
```
中文名: 最小母线电压维持点
寄存器: 0x3E
范围: 3072mV ~ 13312mV (步进 256mV)

【作用】: 当电池欠压或移除时，维持 VSYS 的最低电压

【为什么需要它？】
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
假设你的 MCU 最低工作电压是 3.0V:

  场景: 电池只剩 2.8V 且没有适配器
  
  如果不设 MinSysVolt:
  VSYS = VBAT = 2.8V < 3.0V → MCU 意外复位! 💥
  
  如果设置了 MinSysVolt = 3584mV (3.584V):
  芯片检测到 VBAT 太低
  → 自动断开电池 (BATFET 关闭)
  → 但这样系统也没电了啊?!
  
  所以 MinSysVolt 的真正含义是:
  "如果电池电压低于这个值，且没有适配器，
   那我只能让系统关机了，因为无法维持"

【正确使用策略】:
  设置值应该略高于 MCU 的最低工作电压:
  - CH32L103 最低 2.0V → 设 2560mV 即可
  - 某些 ARM Cortex-M4 需要 3.0V+ → 设 3072mV
  - 一般推荐: 3584mV (留有余量)
```

#### **19. Hi-Z Mode (High Impedance Mode) - 高阻态模式**
```
中文名: 高阻抗模式 / 运输模式 / Ship Mode

【效果】: 让 BQ25710 进入"假死"状态
         静态电流降至 < 1μA (几乎为零!)

【什么时候用它？】
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1️⃣ 产品出厂运输时 (Ship Mode):
   电池在仓库里存放半年
   如果不进入 Hi-Z，芯片自身耗电会把电池彻底放坏
   
   进入方法:
   - EN_HIZ = 1 (Option3[15])
   - 可选: EN_SHIP_DCHG = 1 (同时放电到 3.8V 以下)

2️⃣ 长期闲置存储时:
   设备几个月不用，保存电池寿命

3️⃣ 硬件维修隔离时:
   防止电池通过芯片向外供电，确保维修安全

【唤醒方法】:
  插入 USB/适配器 (VBUS > 3.5V) 或
  按下开机键 (如果有相关电路设计)

【代码示例】:
  // 进入运输模式 (带放电)
  bq25710_enter_hiz_mode(1);  // discharge=1
  
  // 此时电流消耗:
  // 正常模式: ~5mA
  // 低功耗模式: ~50μA  
  // Hi-Z模式:   <1μA  (几乎不耗电!)
```

### 🔄 **六、看门狗与定时器**

#### **20. WDT (Watchdog Timer) - 看门狗定时器**
```
中文名: 看门狗定时器 / 喂狗计时器

【作用】: 防止 MCU 死机后充电失控

【问题场景】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  MCU 正在以 2A 电流快速充电
  突然 MCU 软件跑飞了 (死机/死循环/静电干扰)
  
  如果没有 WDT:
  → 充电永远不停
  → 电池过充 → 鼓包 → 起火 🔥
  
  有了 WDT:
  → BQ25710 内部计时器开始倒计时
  → 如果倒计时结束前没有收到任何 I2C 通信
  → 认为主控已死机
  → 自动将所有寄存器恢复为安全默认值
  → 停止充电 (CHRG_INHIBIT=1)
  → 保住电池性命!

【四种超时选项】:
  Register 0x12 [WDTMR_ADJ]:
  
  ┌──────────┬───────────┬────────────────────────────┐
  │  编码     │  超时时间  │ 适用场景                   │
  ├──────────┼───────────┼────────────────────────────┤
  │  00b      │  禁用     │ 开发调试阶段 (方便单步调试)  │
  │  01b      │  5 秒     │ 需要频繁通信的应用          │
  │  10b      │  88 秒    │ 一般消费电子产品 (推荐)      │
  │  11b      │  175 秒   │ 低功耗长间隔轮询设备        │
  └──────────┴───────────┴────────────────────────────┘

【"喂狗"操作】:
  只要向任意寄存器写入数据 (读不算!)
  看门狗计数器就会清零重新计时
  
  所以在你的主循环里:
  while (1) {
      bq25710_get_charger_status(&status);  // 这只是读
      // 必须有写操作才能喂狗:
      // bq25710_adc_start_single_conversion();  // 写ADC_START位
      
      delay_ms(1000);  // 不要超过 WDT 超时时间!
  }
```

### 🎛️ **七、其他重要术语**

#### **21. NTC (Negative Temperature Coefficient) - 热敏电阻**
```
中文名: 负温度系数热敏电阻
连接引脚: TS (Thermistor Sense)

【作用】: 实时监测电池温度，防止热失控

【工作原理】:
  温度↑ → 电阻↓ (负温度系数)
  
  典型值 (10kΩ @ 25°C 型号):
  ┌──────────┬───────────┐
  │ 温度(°C) │ 电阻值(Ω) │
  ├──────────┼───────────┤
  │   -20    │   67k     │
  │    0     │   33k     │
  │   25     │   10k     │ (标称值)
  │   45     │   3.9k    │
  │   60     │   2.2k    │
  └──────────┴───────────┘

【温度保护策略】(芯片内部自动执行):
  - 冷温 (< 0°C):  停止充电 (锂离子在低温充电会析锂)
  - 低温 (0~10°C): 减半电流
  - 常温 (10~45°C): 正常全速充电
  - 高温 (45~60°C): 降低电流或停止
  - 过热 (> 60°C): 紧急停止充电

【注意】: 必须接 NTC，否则芯片可能拒绝充电!
```

#### **22. PSYS (Power System) - 系统功率监测**
```
中文名: 系统总功率监测引脚
输出: 电流信号 (0.25μA/W 或 1μA/W)

【用途】: 测量当前系统的总功耗

计算公式:
  P_SYS = PSYS引脚电流 / 增益系数
  
  例: 增益 = 0.25μA/W, 测得电流 = 5μA
  → P_SYS = 5μA / 0.25μA/W = 20W

【应用场景】:
  - 笔记本电脑: 监测 CPU+GPU+屏幕总功耗
  - 决定是否需要降频 (PROCHOT 联动)
  - 估算电池续航时间
```

#### **23. CMP (Comparator) - 独立比较器**
```
中文名: 独立模拟比较器
输入: CMPIN 引脚 (外部电压)

【作用】: 用户自定义的电压监测/保护功能

【配置项】:
  Register 0x30:
  - CMP_REF: 内部基准选择 (2.3V 或 1.2V)
  - CMP_POL: 输出极性 (高于阈值=低/高于阈值=高)
  - CMP_DEG: 抗抖动时间 (禁用/1μs/2ms/5s)

【创意用法】:
  1️⃣ 过温保护第二重保障:
     CMPIN 接 NTC 分压 → 温度超限时触发 PROCHOT
  
  2️⃣ 自定义低压锁定:
     CMPIN 接 VSYS 分压 → 电压异常时强制关机
  
  3️⃣ 外部按键检测:
     CMPIN 接按键 → 按下时触发中断唤醒
```

#### **24. PTM (Pass-Through Mode) - 直通模式**
```
中文名: 直通模式 / 透传模式

【特殊功能】: 让 VBUS 直接连通到 SYS (绕过开关调节)

【适用场景】:
  - 需要极低噪声的音频设备供电
  - 测试目的 (排除充电器干扰)
  - 特殊的高速信号传输需求

【代价】: 失去了电压调节能力
  VBUS = VSYS (无法升压降压)
  所以只在特定条件下使用
```

## 2.1 NVDC (Narrow VDC) 电源路径管理

### 架构原理图

```
                        BQ25710 内部结构 (简化)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  VBUS (适配器)                     VSYS (系统负载)
  (5V~20V)                          (3.5V~4.4V)
      │                                   ▲
      │                                   │
      ▼                                   │
  ┌────────┐    BATFET (电池开关)    ┌────────┐
  │ Q1/Q4  ├──┬──────────────────────┤ 系统   │
  │(降压调节)│  │                      │ 输出   │
  └────────┘  │                      └────────┘
              │                           ▲
              │                           │
              ▼                           │
         ┌────────┐                      │
         │ 电池   │──────────────────────┘
         │ (VBAT) │  (当适配器不足时补充)
         └────────┘

【关键特性】:

1️⃣ "即时开机"功能 (Instant-On):
   即使电池完全没电 (0V)，插入适配器后：
   → Q1/Q4 开关管工作在降压模式
   → 直接从 VBUS 给 VSYS 供电
   → MCU 可以立即启动 (不用等电池充到 3V)

2️⃣ 电池独立维护:
   电池可以随时热插拔 (Hot-Swap)
   不影响系统运行

3️⃣ 动态电源分配:
   适配器功率 = 系统功耗 + 电池充电功率
   
   例: 65W 适配器
   ├─ 系统正在跑满载: 消耗 45W
   └─ 剩余 20W 用于充电: ICHG = 20W / 16V(4S) ≈ 1.25A
   → 自动降低充电速度保证系统稳定

4️⃣ 过压保护:
   如果 VBUS > 28V (ACOV 触发):
   → Q1/Q4 立即关断
   → 电池通过 BATFET 继续给系统供电
   → 系统不中断！
```

## 2.2 同步开关整流架构

### 拓扑结构与工作原理

```
BQ25710 采用 **4开关同步 Buck-Boost 变换器**

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

充电模式 (VBUS > VBAT): 降压 (Buck) 工作
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  VBUS (+) ────┬── [Q1] ──┬── [L] ──┬── [Q3] ──┬─── VBAT (+)
               │   (高侧)   │  (电感) │ (低侧同步)│
  VBUS (-) ────┴───────────┴─────────┴──────────┴─── VBAT (-)
  
  工作过程:
  Phase 1: Q1 导通, Q3 关断 → 电感储能 (电流上升)
  Phase 2: Q1 关断, Q3 导通 → 电感续流 (电流下降)
  
  占空比 D = VBAT / VBUS
  
  例: VBUS=12V, VBAT=4.2V(1S) → D = 4.2/12 = 35%


OTG 模式 (VBAT > VBUS): 升压 (Boost) 工作
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  VBAT (+) ────┬── [Q3] ──┬── [L] ──┬── [Q1] ──┬─── VBUS (+)
               │  (低侧)   │  (电感) │ (高侧同步)│
  VBAT (-) ────┴──────────┴─────────┴──────────┴─── VBUS (-)
  
  工作过程:
  Phase 1: Q3 导通, Q1 关断 → 电感从电池取电储能
  Phase 2: Q3 关断, Q1 导通 → 电感能量释放到 VBUS
  
  升压比 M = VBUS / VBAT
  
  例: VBAT=3.7V, VBUS=5V → M = 5/3.7 = 1.35


【同步整流的效率优势】:
传统方案用二极管续流 → 压降 0.3~0.7V → 发热大
BQ25710 用 MOSFET (Q3/Q4) 续流 → 导通电阻 Rds(on) < 20mΩ
→ 压降仅几十 mV → 效率提升 3~5%

实际效率曲线 (以 12V→4S 为例):
  2A 负载: 96%
  4A 负载: 97%
  6A 负载: 96.5% (电感损耗开始占主导)
  8A 负载: 95% (接近极限)
```

### ADC 使用最佳实践

```c
/**
 * @brief 高精度电量计实现 (库仑计数法)
 */

typedef struct {
    int32_t coulomb_count_mAs;   // 库仑累计值 (毫安秒)
    uint16_t design_capacity_mAh; // 设计容量
    uint16_t last_vbat_mV;
    int16_t  last_ichg_mA;
    int16_t  last_idchg_mA;
    uint32_t last_update_tick;
} FuelGauge_t;

static FuelGauge_t s_fuel_gauge = {0};

/**
 * @brief 初始化电量计 (应在系统启动时调用一次)
 */
void fuel_gauge_init(uint16_t capacity_mAh)
{
    memset(&s_fuel_gauge, 0, sizeof(FuelGauge_t));
    s_fuel_gauge.design_capacity_mAh = capacity_mAh;
    
    /* 配置 ADC: 仅开启需要的通道以省电 */
    uint8_t channels = BQ25710_ADC_VBAT | BQ25710_ADC_ICHG | BQ25710_ADC_IDCHG;
    bq25710_adc_configure(channels, 1);  // 连续转换模式
    
    /* 首次读取建立基准 */
    BQ25710_ADC_Result_t adc;
    bq25710_adc_read_all(&adc);
    s_fuel_gauge.last_vbat_mV = adc.vbat_mV;
    s_fuel_gauge.last_ichg_mA = adc.ichg_mA;
    s_fuel_gauge.last_idchg_mA = adc.idchg_mA;
}

/**
 * @brief 更新电量计 (建议每 100ms~1s 调用一次)
 * @return 当前剩余电量百分比 (0~100%)
 */
uint8_t fuel_gauge_update(void)
{
    BQ25710_ADC_Result_t adc;
    if (bq25710_adc_read_all(&adc) != BQ25710_OK) {
        return 0;  // 通信错误
    }
    
    /* 计算时间间隔 (毫秒) */
    uint32_t now = get_system_tick();
    uint32_t delta_ms = now - s_fuel_gauge.last_update_tick;
    s_fuel_gauge.last_update_tick = now;
    
    /* 计算平均电流 (考虑充电和放电) */
    int16_t avg_current = (adc.ichg_mA + s_fuel_gauge.last_ichg_mA) / 2;  // 充电为正
    avg_current -= (adc.idchg_mA + s_fuel_gauge.last_idchg_mA) / 2;       // 放电为负
    
    /* 库仑积分: ΔQ = I × Δt */
    int32_t delta_charge = (int32_t)avg_current * (int32_t)delta_ms;  // 单位: mA·ms
    s_fuel_gauge.coulomb_count_mAs += delta_charge;
    
    /* 保存本次值作为下次参考 */
    s_fuel_gauge.last_vbat_mV = adc.vbat_mV;
    s_fuel_gauge.last_ichg_mA = adc.ichg_mA;
    s_fuel_gauge.last_idchg_mA = adc.idchg_mA;
    
    /* 转换为百分比 */
    int32_t total_capacity_mAs = (int32_t)s_fuel_gauge.design_capacity_mAh * 3600;
    int32_t percent = (s_fuel_gauge.coulomb_count_mAs * 100) / total_capacity_mAs;
    
    /* 限幅 */
    if (percent > 100) percent = 100;
    if (percent < 0) percent = 0;
    
    return (uint8_t)percent;
}

/**
 * @brief 校准电量计 (当已知电池状态时调用)
 * @param percent: 当前的真实电量百分比
 */
void fuel_gauge_calibrate(uint8_t percent)
{
    int32_t total = (int32_t)s_fuel_gauge.design_capacity_mAh * 3600;
    s_fuel_gauge.coulomb_count_mAs = (int32_t)percent * total / 100;
}
```

```c
/**
 * @brief PDO (Power Data Object) 结构定义
 */
typedef struct {
    uint8_t  pdo_type;       // 0=Fixed, 1=Battery, 2=Variable
    uint16_t voltage_mv;     // 电压 (mV)
    uint16_t max_current_ma; // 最大电流 (mA)
    uint16_t max_power_mw;   // 最大功率 (mW)
} PDO_Profile_t;

/* 常见 PD 3.0 档位表 */
static const PDO_Profile_t pd_profiles[] = {
    { .pdo_type = 0, .voltage_mv = 5000,  .max_current_ma = 3000, .max_power_mw = 15000 },  // Profile 1: 5V/3A
    { .pdo_type = 0, .voltage_mv = 9000,  .max_current_ma = 3000, .max_power_mw = 27000 },  // Profile 2: 9V/3A
    { .pdo_type = 0, .voltage_mv = 12000, .max_current_ma = 3000, .max_power_mw = 36000 },  // Profile 3: 12V/3A
    { .pdo_type = 0, .voltage_mv = 15000, .max_current_ma = 3000, .max_power_mw = 45000 },  // Profile 4: 15V/3A
    { .pdo_type = 0, .voltage_mv = 20000, .max_current_ma = 3250, .max_power_mw = 65000 },  // Profile 5: 20V/3.25A
};

/**
 * @brief PD 协商成功后更新 BQ25710 参数
 * @param pdo_index: 选中的 PDO 档位索引 (0~4)
 */
void pd_negotiation_success(uint8_t pdo_index)
{
    if (pdo_index >= sizeof(pd_profiles)/sizeof(PDO_Profile_t)) {
        return;
    }
    
    const PDO_Profile_t *pdo = &pd_profiles[pdo_index];
    
    printf("[PD] Negotiated: %dmV @ %dma (%dmW)\n",
           pdo->voltage_mv, pdo->max_current_ma, pdo->max_power_mw);
    
    /* 步骤1: 设置 VINDPM (留 10% 余量防止拉死适配器) */
    uint16_t vindpm = (pdo->voltage_mv * 90) / 100;
    vindpm = (vindpm / 64) * 64;  // 对齐 64mV 步进
    bq25710_set_vindpm_threshold(vindpm);
    
    /* 步骤2: 设置 IIN_HOST (留 5% 余量) */
    uint16_t iin_host = (pdo->max_current_ma * 95) / 100;
    iin_host = (iin_host / 50) * 50;  // 对齐 50mA 步进
    bq25710_set_input_current_limit(iin_host);
    
    /* 步骤3: 启动 ICO 重新测量实际能力 (推荐) */
    bq25710_enable_ico(1);
    
    printf("[PD] Applied: VINDPM=%dmV, IIN=%dmA\n", vindpm, iin_host);
}

/**
 * @brief 处理 PD PR_Swap (Power Role Swap) 请求
 * 即从"受电方"(Sink)切换为"供电方"(Source)
 */
void handle_pd_pr_swap_to_source(void)
{
    printf("[PD] Switching to Source role (OTG mode)...\n");
    
    /* 1. 先停止充电 */
    bq25710_disable_charging();
    
    /* 2. 配置 OTG 参数 (通常固定 5V/2A 或 9V/2A) */
    uint16_t otg_voltage = 5000;   // 标准 USB 电压
    uint16_t otg_current = 2100;   // USB 3.0 最大电流
    
    /* 3. 使能 OTG */
    int ret = bq25710_enable_otg(otg_voltage, otg_current);
    if (ret != BQ25710_OK) {
        printf("[PD] ERROR: Failed to enable OTG (%d)\n", ret);
        return;
    }
    
    printf("[PD] Now acting as power source: %dV / %.1fA\n",
           otg_voltage/1000, otg_current/1000.0);
}
```

### SYS_SHORT (系统短路) 特殊保护

```
这是最严重的故障之一，有特殊的多次重试机制:

正常情况: VSYS > 2.4V ✓
异常情况: VSYS < 2.4V (可能短路!)

BQ25710 的自动恢复流程:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

第1次检测到短路:
  → 尝试重启 (关断再重新开启 BATFET)
  → 等待 100ms 看 VSYS 是否恢复
  → 如果恢复 → 继续正常工作 (计数器=1)

第2次短路:
  → 再次尝试重启
  → 如果又失败 → 计数器=2

... (最多尝试 7 次) ...

第7次仍然失败:
  → 设置 SYS_SHORT_FAULT = 1 (永久锁存!)
  → BATFET 永久关闭 (电池彻底隔离)
  → 只能通过写寄存器清除 (且短路必须先排除)

为什么设计成 7 次?
  → 有些短路是瞬时的 (如插拔外设的瞬间)
  → 给系统自我恢复的机会
  → 但如果是真短路，反复尝试只会损坏器件
  → 7次是一个经验平衡点

【代码中的处理】:
int ret = bq25710_get_fault_status(&fault_mask);
if (fault_mask & (1 << 3)) {  // SYS_SHORT 位
    printf("FATAL: System short circuit after 7 retries!\n");
    printf("Action: Check for hardware short, then call clear_faults()\n");
    // 此时不要盲目 clear_faults(), 应该先排查硬件!
}
```

### 案例 1: I2C 通信失败

```
症状: bq25710_init() 返回 ERR_I2C_NACK 或 ERR_DEVICE_ID

诊断步骤:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Step 1: 硬件连接检查
  ☐ SDA/SCL 是否接对? (PB7/PB6)
  ☐ 上拉电阻是否存在? (4.7kΩ 到 3.3V)
  ☐ ADDR 引脚电平? (GND→0x6B, VTS→0x09)
  ☐ VDD 是否有电? (3.3V 或 5V)
  ☐ GND 是否共地?

Step 2: I2C 总线扫描
  用示波器或逻辑分析仪观察:
  - 应该看到 Start 条件 → 地址 → ACK/NACK → 数据 → Stop
  - 如果完全没有波形 → MCU I2C 外设未初始化
  - 如果有波形但 NACK → 地址错误或器件未上电

Step 3: 地址验证
  尝试两个常见地址:
  ├── 0x6B (ADDR=GND, 最常用)
  └── 0x09 (ADDR=接TS或VTS)

Step 4: 波形质量检查
  ☐ 上升沿是否够陡? (应 <100ns, 上拉电阻不能太大)
  ☐ 是否有过冲/振铃? (可能需要串接 33Ω 电阻)
  ☐ 总线频率? (BQ25710 支持 100kHz 和 400kHz)

Step 5: 软件层面
  ☐ i2c_if.read/write 函数指针是否有效?
  ☐ 是否在调用 API 前调用了 init()?
  ☐ 其他 I2C 设备是否能正常通信?
```

### 案例 2: 充电但不进电流

```
症状: 插入适配器后 CHRG_OK=1, 但 ICHG=0

可能原因及解决方案:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

原因 A: CHRG_INHIBIT 被置位 (最常见!)
  检查方法: 读取 ChargeOption0 寄存器, 看 bit[0]
  解决: 调用 bq25710_enable_charging()

原因 B: 电池电压已经充满 (≥ VREG)
  检查方法: 读 ADC_VBAT, 如果 ≥ 4190mV 则正常
  解决: 这是正常的恒压截止阶段

原因 C: 电池温度超出范围
  检查方法: 测量 TS 引脚电压
  - 正常: 0.55V~2.35V (对应 -20°C ~ 60°C)
  - 如果 TS 开路或短路: 芯片会禁止充电
  解决: 检查 NTC 热敏电阻焊接和选型

原因 D: 输入功率全部被系统消耗了
  场景: 5V/2A 适配器, 系统本身就要 1.8A
  结果: 剩余 0.2A 可能低于最小充电电流
  检查: 读 IIN_DPM, 看是否 < 200mA
  解决: 换更大功率的适配器, 或降低系统功耗

原因 E: Watchdog 超时复位了寄存器
  检查: ChargeOption0 的 WDTMR_ADJ 字段
  如果不是 DISABLE, 且长时间没有 I2C 通信
  → 所有寄存器恢复默认值 (包括 CHRG_INHIBIT=0 是默认OK的)
  但其他自定义设置会丢失
  解决: 要么禁用 WDT, 要么定期"喂狗"
```

### 案例 3: OTG 无输出

```
症状: 调用 enable_otg() 后 VBUS 仍为 0V

检查清单:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

□ 1. 电池电压是否足够?
   OTG 要求 VBAT > 3.0V (低量程) 或 > 3.5V (高量程)
   如果电池没电, 无法反向供电!

□ 2. EN_OTG 位是否真的写入了?
   读回 ChargeOption3[12], 确认 = 1

□ 3. OTG 电压是否在允许范围内?
   高量程: 4.28V ~ 20.8V
   低量程: 3.0V ~ 19.52V
   如果设置了 4.1V 且在高量程模式 → 会被拒绝!

□ 4. 是否有 OTG_FAULT?
   读 ChargerStatus[0] (OTG_UVP) 和 [1] (OTG_OVP)
   如果置位 → 先 clear_faults() 再重试

□ 5. 负载是否过重?
   OTG_CURRENT 设置是否满足负载需求?
   如果负载要 2A 但限流设了 500mA → 输出会被拉低

□ 6. 外部电路是否有问题?
   - VBUS 引脚到 USB-C 口之间是否有开关/MOS?
   - 该开关是否被正确打开?
   - 是否有保险丝熔断?

调试技巧: 用万用表测 OTG_Voltage 寄存器对应的实际引脚电压
  如果引脚有电压但 USB-C 口没有 → 问题在外部电路
  如果引脚也没电压 → 问题在芯片配置
```

```
1. VIN 电容: 至少 2 个, 一个大的 (22μF) + 一个小的 (100nF)
   → 大电容滤低频, 小电容滤高频
   
2. 电感选择:
   - 饱和电流 > 最大充电电流 × 1.3
   - 直流电阻 (DCR) < 20mΩ (影响效率)
   - 推荐型号: Coilcraft XAL系列, Murata LQH系列
   
3. SW 节点走线:
   - 不要在 SW 下走其他信号线!
   - SW 铜箔面积适中 (太大会辐射EMI, 太小会发热)
   - 推荐: 用多过孔连到底层地 (减小回路面积)
   
4. 散热:
   - 芯片底部裸焊盘 (Exposed Pad) 必须焊接到地!
   - 打至少 5个过孔连接到内层地平面
   - 如果长期大电流工作, 考虑加散热片
```

## 8.1 效率优化到极致

```c
/**
 * @brief 将充电效率推向理论极限 (目标 > 97%)
 */

void ultimate_efficiency_tuning(void)
{
    /* 技巧1: 根据输入输出电压差动态调整开关频率 */
    /*
     * 压差越大 → 需要更低的频率来降低开关损耗
     * 压差越小 → 可以提高频率减小电感体积
     */
    uint16_t vbus, vbat;
    bq25710_adc_read_channel(BQ25710_ADC_VBUS, &vbus);
    bq25710_adc_read_channel(BQ25710_ADC_VBAT, &vbat);
    
    uint16_t delta_v = (vbus > vbat) ? (vbus - vbat) : (vbat - vbus);
    
    if (delta_v > 15000) {       // 压差 > 15V
        bq25710_set_switch_frequency(BQ25710_FREQ_800KHZ);  // 低频
    } else if (delta_v > 5000) {  // 压差 5~15V
        bq25710_set_switch_frequency(BQ25710_FREQ_800KHZ);  // 保持低频
    } else {                     // 压差 < 5V
        // 可以保持 800kHz 或适当提高
    }
    
    /* 技巧2: 在轻载时进入 PFM 模式 (如果固件支持) */
    /*
     * BQ25710 本身不直接支持 PFM, 但可以通过降低充电电流
     * 让芯片自然进入轻载高效区
     * 
     * 轻载定义为: ICHG < 10% of ICHG_max
     * 此时固定频率PWM的固定损耗占比变大
     */
    
    /* 技巧3: 优化死区时间 (Dead Time) */
    /*
     * Q1和Q3不能同时导通 (否则直通短路!)
     * 但也不能都关断太久 (否则电感电流断流, 效率下降)
     * 
     * BQ25710 的死区时间是自适应的 (内部优化)
     * 我们只需要确保:
     * - 电感不要饱和 (选对Isat)
     * - 不要工作在断续导通模式 (DCM) - 保持 CCM
     */
    
    /* 技巧4: 减小检测电阻损耗 */
    /*
     * 10mΩ 电阻在 6A 时功耗: I²R = 36×0.01 = 0.36W (纯损耗!)
     * 改用 5mΩ: 损耗降至 0.09W (节省 0.27W)
     * 
     * 但代价: ADC 分辨率减半 (LSB 从 25mA 变 50mA)
     * 权衡: 对于不需要精确电流测量的应用, 值得!
     */
}
```

## 9.1 产线自动化测试流程

```c
/**
 * @brief 产线 ATE (Automatic Test Equipment) 测试程序
 * 用于批量生产时的快速检验
 */

typedef enum {
    TEST_PASS = 0,
    TEST_FAIL_I2C,
    TEST_FAIL_REG,
    TEST_FAIL_CHARGE,
    TEST_FAIL_OTG,
    TEST_FAIL_PROTECT,
} TestResult_t;

TestResult_t production_line_test(void)
{
    TestResult_t result = TEST_PASS;
    uint16_t read_val;
    
    /* ===== Test 1: I2C 通信 ===== */
    if (bq25710_check_device_id() != BQ25710_OK) {
        return TEST_FAIL_I2C;
    }
    
    /* ===== Test 2: 寄存器读写回环 ===== */
    /* 写一个特殊值, 读回来看是否一致 */
    uint16_t test_pattern = 0xA5A5;
    bq25710_write_reg16(0x14, test_pattern);  // ChargeCurrent (无害寄存器)
    bq25710_read_reg16(0x14, &read_val);
    if (read_val != test_pattern) {
        result = TEST_FAIL_REG;
        goto restore_defaults;
    }
    
    /* ===== Test 3: 充电功能基本检查 ===== */
    bq25710_reset();  // 先复位确保干净状态
    
    /* 设置一个小电流并验证 */
    bq25710_set_charge_current(512);  // 512mA
    bq25710_get_charge_current(&read_val);
    if (read_val != 512) {
        result = TEST_FAIL_CHARGE;
        goto restore_defaults;
    }
    
    /* 检查充电使能 */
    bq25710_enable_charging();
    BQ25710_ChargeOption0_t opt0;
    bq25710_read_reg16(0x12, &opt0.all);
    if (opt0.bits.CHRG_INHIBIT != 0) {
        result = TEST_FAIL_CHARGE;
        goto restore_defaults;
    }
    
    /* ===== Test 4: OTG 功能基本检查 (如果有条件) ===== */
    /* 注意: 此测试需要电池电压 > 3V, 否则跳过 */
    uint16_t vbat;
    if (bq25710_adc_read_channel(BQ25710_ADC_VBAT, &vbat) == BQ25710_OK && vbat > 3000) {
        int ret = bq25710_enable_otg(5000, 500);  // 5V/0.5A (安全值)
        if (ret != BQ25710_OK) {
            result = TEST_FAIL_OTG;
            goto restore_defaults;
        }
        
        /* 验证 OTG 状态位 */
        BQ25710_ChargerStatus_t status;
        bq25710_get_charger_status(&status);
        if (!status.bits.IN_OTG) {
            result = TEST_FAIL_OTG;
            bq25710_disable_otg();
            goto restore_defaults;
        }
        
        bq25710_disable_otg();  // 恢复
    }
    
restore_defaults:
    /* 恢复出厂设置 */
    bq25710_reset();
    
    return result;
}
```

## 🔋 **特性一：从电池上电 (Battery-Only Power Up)**

### 📌 **核心概念**

```
场景: 设备没有连接充电器，仅靠电池供电时的启动过程

【关键机制】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 启动条件: VBAT > VVBAT_UVLOZ (电池欠压锁定阈值)
   - 如果电池电压太低 (如 < 2.5V), 芯片不会启动
   - 这是保护深度放电电池的措施

2. BATFET 自动导通:
   ┌─────────────────────────────────────┐
   │  电池 (+) ──→ [BATFET] ──→ 系统    │
   │                  ↓                │
   │            (导通状态)              │
   │             直接连通              │
   └─────────────────────────────────────┘

3. 默认低功耗模式:
   - REG0x12[15] = EN_LWPWR = 1 (默认)
   - 静态电流极低 (~150μA 或更低)
   - LDO 保持关闭 (节省功耗)
   - IBAT/PSYS 缓冲器关闭

4. 进入性能模式后的操作:
   当 MCU 设置 REG0x12[15] = 0 时:
   ├── 可启用 IBAT 缓冲器 → 监测放电电流
   ├── 可启用 PSYS → 监测系统功率
   ├── 可启用 PROCHOT → 过载报警
   └── 可启用独立比较器 → 自定义保护
```

### ⚙️ **实际应用代码**

```c
/**
 * @brief 纯电池模式下的初始化序列
 */
void battery_only_power_up_sequence(void)
{
    /* 步骤1: 检查电池电压是否足够启动 */
    uint16_t vbat;
    bq25710_adc_read_channel(BQ25710_ADC_VBAT, &vbat);
    
    if (vbat < 2500) {  // 假设 UVLOZ 约 2.5V
        printf("[ERROR] Battery voltage too low for startup: %dmV\n", vbat);
        /* 此时芯片可能根本没工作, 无法通信! */
        return;
    }
    
    printf("[INFO] Battery voltage OK: %dmV, starting up...\n", vbat);
    
    /* 步骤2: 芯片应该已经自动导通了 BATFET */
    /* 我们可以验证系统是否有电 */
    uint16_t vsys;
    bq25710_adc_read_channel(BQ25710_ADC_VSYS, &vsys);
    
    /* 在纯电池模式下, VSYS 应该 ≈ VBAT (减去 BATFET 的压降 ~50mV) */
    if ((vbat > vsys) && ((vbat - vsys) < 200)) {
        printf("[OK] BATFET is conducting normally\n");
        printf("     VBAT=%dmV, VSYS=%dmV, Delta=%dmV\n", 
               vbat, vsys, vbat - vsys);
    }
    
    /* 步骤3: 从低功耗切换到性能模式 (如果需要 ADC) */
    bq25710_set_low_power_mode(0);  // EN_LWPWR = 0
    
    /* 步骤4: 使能需要的监测功能 */
    BQ25710_ChargeOption1_t opt1;
    bq25710_read_reg16(BQ25710_REG_CHARGE_OPTION_1, &opt1.all);
    
    opt1.bits.EN_IBAT = 1;   // 使能电池电流监测
    opt1.bits.EN_PSYS = 1;   // 使能功率监测 (可选, 增加约 100μA)
    bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_1, opt1.all);
    
    /* 步骤5: 配置 ADC 开始监测 */
    uint8_t adc_channels = BQ25710_ADC_VBAT | BQ25710_ADC_VSYS | 
                          BQ25710_ADC_IDCHG;  // 放电电流
    bq25710_adc_configure(adc_channels, 1);  // 连续模式
    
    printf("[SUCCESS] Battery-only mode initialized\n");
}
```

## ⚡ **特性二：VAP 模式 (Voltage Adaptive Peak / 涡轮加速)**

### 📌 **这是什么？**

```
VAP = Voltage Adaptive Peak (自适应峰值电压模式)

【解决的核心问题】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

想象一台高性能笔记本电脑:

  正常功耗: 15W (CPU基频运行)
  涡轮加速:  100W (CPU短时间全速, 如编译、渲染)

问题来了:
  充电器只能提供 65W
  电池可以提供 50W 瞬时放电
  
  当 CPU 需要 100W 时:
  65W (充电器) + ?W (电池) = 100W
  
  传统方案: 电池不够用, 只能降频, 性能暴跌!

VAP 方案的巧妙之处:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  在 CPU 不忙的时候 (功耗 20W):
  ┌────────────────────────────────────────┐
  │  充电器(65W)                            │
  │     ├─→ 系统(20W)                       │
  │     └─→ 多余的 45W 存到哪里?            │
  │         └─→ 给 VBUS 输入电容"充电"到 20V!│  ← 关键!
  └────────────────────────────────────────┘
  
  VBUS 电容就像一个"能量蓄水池"
  平时慢慢蓄水 (充电到高压), 急用时大量放水

  当 CPU 涡轮加速 (需要 100W, 持续 50ms):
  ┌────────────────────────────────────────┐
  │  能量来源:                              │
  │  ├─ 充电器提供: 65W                    │
  │  ├─ VBUS电容释放: 20W (电压从20V降到12V) │
  │  └─ 电池补充: 15W                      │
  │                                         │
  │  总计: 100W ✓ 满足需求!               │
  │  CPU全速运行, 不降频!                 │
  └────────────────────────────────────────┘

  这就是为什么叫 "涡轮加速" - 短时间突破 TDP 限制!
```

### ⚙️ **VAP 配置步骤详解**

```c
/**
 * @brief 配置 VAP (Voltage Adaptive Peak) 模式
 * 用于支持 CPU 涡轮加速等高瞬态功耗场景
 * 
 * @param vap_target_vbus_mv: VBUS 电容充电目标电压 (建议 15V~20V)
 * @param vap_current_limit_ma: VBUS 放电电流限制
 * @param vsys_min_mV: 系统最低维持电压
 * @param vsys_throttle_mV: 触发 PROCHOT 节流的 VSYS 阈值
 * @param vsys_prochot_mV: 触发 PROCHOT 报警的 VSYS 阈值
 */
void configure_vap_mode(uint16_t vap_target_vbus_mv,
                       uint16_t vap_current_limit_ma,
                       uint16_t vsys_min_mV,
                       uint16_t vsys_throttle_mV,
                       uint16_t vsys_prochot_mV)
{
    printf("[VAP] Configuring Voltage Adaptive Peak mode...\n");
    
    /*
     * 步骤1: 设置 OTG Voltage 寄存器 (0x3B) 作为 VBUS 目标电压
     * 注意: 在 VAP 模式下, 这个寄存器的含义变了!
     *       它不再是 OTG 输出电压, 而是 VBUS 电容的目标充电电压
     */
    bq25710_set_otg_voltage(vap_target_vbus_mv);
    printf("[VAP] Step1: VBUS target voltage = %dmV\n", vap_target_vbus_mv);
    
    /*
     * 步骤2: 设置电流限制
     * 2a. OTG Current (0x3C): VBUS 放电限流
     */
    bq25710_set_otg_current_limit(vap_current_limit_ma);
    
    /*
     * 2b. ProchotOption1 IDCHG 阈值 (0x34[15:10]): 
     *     电池补充电流阈值 (当系统从 VBUS+电池 取电时)
     */
    BQ25710_ProchotOption1_t prochot1;
    bq25710_read_reg16(BQ25710_REG_PROCHOT_OPTION_1, &prochot1.all);
    
    /* IDCHG_VTH 计算: 范围 0~32256mA, 步进 512mA */
    /* 设置为 4096mA (8A) 作为示例 */
    uint16_t idchg_threshold = 8000;  // 8A
    prochot1.bits.IDCHG_VTH = idchg_threshold / 512;
    
    bq25710_write_reg16(BQ25710_REG_PROCHOT_OPTION_1, prochot1.all);
    printf("[VAP] Step2: Current limits set (OTG=%dmA, IDCHG_th=%dmA)\n",
           vap_current_limit_ma, idchg_threshold);
    
    /*
     * 步骤3: 设置 MinSystemVoltage (0x3E[13:8])
     * 当 VBUS 电容向系统放电时, VSYS 不能低于此值
     * 否则触发电池补充
     */
    bq25710_set_min_system_voltage(vsys_min_mV);
    printf("[VAP] Step3: Min system voltage = %dmV\n", vsys_min_mV);
    
    /*
     * 步骤4: 配置 PROCHOT VSYS 阈值 (寄存器 0x33)
     * 4a. TH1 (bits[7:4]): 触发 VAP 将 VBUS 放电的阈值
     *     当 VSYS 低于此值时, 开始从 VBUS 电容取电
     *     
     * 4b. TH2 (bits[3:2]): 触发节流 SoC 的阈值
     *     当 VSYS 低于此值时, PROCHOT 有效, CPU 必须降频
     */
    BQ25710_ProchotOption0_t prochot0;
    bq25710_read_reg16(BQ25710_REG_PROCHOT_OPTION_0, &prochot0.all);
    
    /* VSYS_TH1 编码: 
     * 0000=5.9V, 0001=6.0V, ..., 1111=7.4V (步进 0.1V)
     * 公式: value = (voltage_mV - 5900) / 100
     */
    uint8_t th1_code = (vsys_throttle_mV - 5900) / 100;
    if (th1_code > 15) th1_code = 15;
    prochot0.bits.VSYS_TH1 = th1_code;
    
    /* VSYS_TH2 编码:
     * 00=5.9V, 01=6.2V, 10=6.5V, 11=6.8V
     */
    uint8_t th2_code;
    if (vsys_prochot_mV <= 6200) th2_code = 0b01;      // 6.2V
    else if (vsys_prochot_mV <= 6500) th2_code = 0b10;  // 6.5V
    else th2_code = 0b11;                               // 6.8V
    prochot0.bits.VSYS_TH2 = th2_code;
    
    bq25710_write_reg16(BQ25710_REG_PROCHOT_OPTION_0, prochot0.all);
    printf("[VAP] Step4: VSYS thresholds (TH1=%dmV, TH2=%dmV)\n",
           5900 + th1_code * 100, 
           6200 + (th2_code - 1) * 300);
    
    /*
     * 步骤5: 启动 VAP 模式!
     * 关键配置:
     * - REG0x32[5] (EN_CON_VAP) = 0: 启用保守 VAP 模式
     * - REG0x32[12] (EN_OTG) = 0: 不使用标准 OTG
     * - OTG/VAP 引脚拉高: 硬件使能信号
     */
    BQ25710_ChargeOption3_t opt3;
    bq25710_read_reg16(BQ25710_REG_CHARGE_OPTION_3, &opt3.all);
    
    opt3.bits.EN_CON_VAP = 0;   // 启用 VAP
    opt3.bits.EN_OTG = 0;      // 禁止标准 OTG
    opt3.bits.OTG_VAP_MODE = 0; // 引脚控制 VAP 模式 (不是 OTG)
    
    bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_3, opt3.all);
    
    /* 注意: 还需要将 OTG/VAP 物理引脚拉高! 
     * 这通常通过 GPIO 控制, 或者硬件设计时接固定高电平 */
    // set_vap_pin_high();  // 平台相关代码
    
    printf("[VAP] Step5: VAP mode ENABLED!\n");
    printf("[VAP] ============================================\n");
    printf("[VAP] Target VBUS:    %dV\n", vap_target_vbus_mv / 1000);
    printf("[VAP] Max discharge:  %.1fA\n", vap_current_limit_ma / 1000.0);
    printf("[VAP] Min VSYS:      %dV\n", vsys_min_mV / 1000);
    printf("[VAP] Throttle at:    %dV\n", vsys_throttle_mV / 1000);
    printf("[VAP] PROCHOT at:     %dV\n", vsys_prochot_mV / 1000);
    printf("[VAP] ============================================\n");
}

/**
 * @brief 退出 VAP 模式
 */
void exit_vap_mode(void)
{
    /* 方法1: 写入 EN_CON_VAP = 1 */
    BQ25710_ChargeOption3_t opt3;
    bq25710_read_reg16(BQ25710_REG_CHARGE_OPTION_3, &opt3.all);
    opt3.bits.EN_CON_VAP = 1;  // 退出 VAP
    bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_3, opt3.all);
    
    /* 方法2: 将 OTG/VAP 引脚拉低 */
```

### ⚙️ **关键参数: CHRG_OK 指示器**

```c
/**
 * @brief CHRG_OK 引脚状态监控
 * 
 * CHRG_OK 是开漏输出 (Open-Drain), 需要外部上拉电阻
 * 高电平有效 = 一切正常
 * 低电平 = 有问题!
 */

typedef enum {
    CHRG_OK_GOOD = 0,      // 正常
    CHRG_OK_LOW_VBUS,      // VBUS 不足 (未插好或线太长)
    CHRG_OK_OVERVOLT,      // VBUS 过压 (>28V)
    CHRG_OK_MOSFET_FAIL,   // 开关管故障
    CHRG_OK_INDUCTOR_FAIL,  // 电感故障/短路
    CHRG_OK_THERMAL_SHUTDOWN  // 热关断
} CHRG_OK_Status_t;

/**
 * @brief 检查 CHRG_OK 状态并诊断原因
 */
CHRG_OK_Status_t check_chrg_ok_status(void)
{
    /* 方法1: 通过引脚直接读取 (如果有 GPIO 连接) */
    // uint8_t pin_state = GPIO_ReadPin(CHRG_OK_GPIO_PORT, CHRG_OK_PIN);
    // return (pin_state == 1) ? CHRG_OK_GOOD : CHRG_OK_UNKNOWN;
    
    /* 方法2: 通过寄存器间接推断 */
    BQ25710_ChargerStatus_t status;
    bq25710_get_charger_status(&status);
    
    /* 检查 AC_STAT (适配器是否存在) */
    if (!status.bits.AC_STAT) {
        return CHRG_OK_LOW_VBUS;  // 无适配器
    }
    
    /* 检查故障标志 */
    uint16_t fault_mask;
    bq25710_get_fault_status(&fault_mask);
    
    if (fault_mask & (1 << 7)) {  // ACOV 位
        return CHRG_OK_OVERVOLT;
    }
    
    if (fault_mask & 0x03) {  // OTG_UVP or OTG_OVP
        return CHRG_OK_MOSFET_FAIL;
    }
    
    /* 如果所有检查都通过, CHRG_OK 应该为高 */
    return CHRG_OK_GOOD;
}

void print_chrg_ok_diagnosis(void)
{
    CHRG_OK_Status_t st = check_chrg_ok_status();
    
    const char *messages[] = {
        "[OK] Charging status normal - adapter connected and healthy",
        "[WARN] CHRG_OK LOW: VBUS below threshold - check adapter connection",
        "[ERROR] CHRG_OK LOW: Overvoltage detected (>28V) - remove adapter immediately!",
        "[FAULT] CHRG_OK LOW: MOSFET or inductor failure detected",
        "[FAULT] CHRG_OK LOW: Thermal shutdown - let it cool down!"
    };
    
    printf("%s\n", messages[st]);
    
    /* 额外信息: 读取当前 VBUS 电压确认 */
    uint16_t vbus;
    bq25710_adc_read_channel(BQ25710_ADC_VBUS, &vbus);
    printf("     VBUS voltage: %d.%02dV\n", vbus/1000, (vbus%1000)/10);
}
```

```
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
```

## 🎛️ **特性五：Hi-Z 高阻态模式 (Shipping Mode)**

### 📌 **进入条件与效果**

```

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

方式1: 硬件触发
  ILIM_HIZ 引脚电压 < 0.4V (被拉低)
  
方式2: 软件触发
  写入 REG0x32[15] (EN_HIZ) = 1

【Hi-Z 模式的特征】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✅ 极低的静态电流 (< 1μA, 典型值 0.5μA)

   - 电池自放电成为主要耗电来源

✅ 输入源仍可存在 (不同于完全关机)
   - 如果插着充电器, 维持极小电流取电
   - REGN LDO 保持工作 (用于唤醒检测)

✅ BATFET 关断
   - 电池与系统完全隔离
   - 即使电池漏电也不会影响待机

⚠️ 大部分功能禁用
   - 充电停止
   - ADC 关闭
   - I2C 仍然可通信 (但功能受限)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

           低功耗模式 (LWPWR)      Hi-Z 模式
           ─────────────────      ──────────

充电功能:   可用 (受限)              禁止
```

### ⚙️ **运输模式完整实现**

```c
/**
 * @brief 进入 Hi-Z 运输模式
 * @param discharge_to_ship: 是否先放电到 3.8V 以下
 *        true  -> 先放电再进入 Hi-Z (推荐用于出厂)
 *        false -> 直接进入 Hi-Z (保留电池电量)
 */
int enter_shipping_mode(uint8_t discharge_to_ship)
{
    int ret;
    
    printf("[SHIP] Entering Hi-Z shipping mode...\n");
    printf("[SHIP] Discharge option: %s\n", 
           discharge_to_ship ? "ENABLED (will discharge to <3.8V)" : "DISABLED");
    
    /* ===== 步骤1: 停止所有活动功能 ===== */
    
    /* 1.1 停止充电 */
    ret = bq25710_disable_charging();
    if (ret != BQ25710_OK) return ret;
    
    /* 1.2 确保 OTG 已关闭 */
    BQ25710_ChargeOption3_t opt3;
    ret = bq25710_read_reg16(BQ25710_REG_CHARGE_OPTION_3, &opt3.all);
    if (ret != BQ25710_OK) return ret;
    
    if (opt3.bits.EN_OTG) {
        ret = bq25710_disable_otg();
        if (ret != BQ25710_OK) return ret;
    }
    
    /* 1.3 关闭 ADC 以省电 */
    bq25710_adc_configure(0, 0);  // 全部通道关闭
    
    /* 1.4 关闭 PSYS 和 IBAT 缓冲器 */
    BQ25710_ChargeOption1_t opt1;
    ret = bq25710_read_reg16(BQ25710_REG_CHARGE_OPTION_1, &opt1.all);
    if (ret != BQ25710_OK) return ret;
    
    opt1.bits.EN_PSYS = 0;
    opt1.bits.EN_IBAT = 0;
    ret = bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_1, opt1.all);
    if (ret != BQ25710_OK) return ret;
    
    /* ===== 步骤2: 配置运输放电 (可选) ===== */
    if (discharge_to_ship) {
        /* 设置 EN_SHIP_DCHG = 1 */
        /* 这会让芯片在 SRN 引脚产生 140ms 的放电脉冲 */
        /* 将电池放电到约 3.8V (安全存储电压) */
        
        opt1.bits.EN_SHIP_DCHG = 1;
        ret = bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_1, opt1.all);
        if (ret != BQ25710_OK) return ret;
        
        printf("[SHIP] Ship discharge enabled - battery will drain to ~3.8V\n");
        printf("[SHIP] This may take several minutes/hours depending on capacity\n");
    } else {
        opt1.bits.EN_SHIP_DCHG = 0;
        ret = bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_1, opt1.all);
        if (ret != BQ25710_OK) return ret;
    }
    
    /* ===== 步骤3: 进入 Hi-Z 模式 ===== */
    
    /* 设置 EN_HIZ = 1 */
    opt3.bits.EN_HIZ = 1;
    ret = bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_3, opt3.all);
    if (ret != BQ25710_OK) return ret;
    
    /* 可选: 同时将 ILIM_HIZ 引脚拉低 (如果硬件支持) */
    // set_ilim_hiz_low();  // GPIO 操作
    
    printf("[SHIP] Hi-Z mode entered successfully!\n");
    printf("[SHIP] Current consumption: < 1μA (typical 0.5μA)\n");
    printf("[SHIP] To wake up: Connect adapter or write EN_HIZ=0\n");
    
    return BQ25710_OK;
}

/**
 * @brief 从 Hi-Z 模式唤醒
 */
int wake_from_hiz_mode(void)
{
    int ret;
    
    printf("[WAKE] Waking from Hi-Z shipping mode...\n");
    
    /* 方法1: 清除 EN_HIZ 位 */
    BQ25710_ChargeOption3_t opt3;
    ret = bq25710_read_reg16(BQ25710_REG_CHARGE_OPTION_3, &opt3.all);
    if (ret != BQ25710_OK) return ret;
    
    opt3.bits.EN_HIZ = 0;
    ret = bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_3, opt3.all);
    if (ret != BQ25710_OK) return ret;
    
    /* 方法2: 将 ILIM_HIZ 引脚拉高 (如果之前拉低了) */
    // set_ilim_hiz_high();
    
    /* 等待芯片退出 Hi-Z (可能需要几毫秒) */
    delay_ms(10);
    
    /* 重新初始化配置 */
    ret = bq25710_init_default();
    if (ret != BQ25710_OK) return ret;
    
    printf("[WAKE] Successfully exited Hi-Z mode!\n");
    printf("[WAKE] Driver reinitialized with default settings\n");
    
    return BQ25710_OK;
}
```

## 🔧 **特性六：电感值自动检测 (IADPT)**

### 📌 **为什么需要检测电感？**

```
BQ25710 需要知道用了多大的电感, 因为:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 优化开关时序:
   - 小电感 (1μH): 电流变化快 → 需要更短的开关时间
   - 大电感 (4.7μH): 电流变化慢 → 可以用较长的开关时间
   
2. 峰值电流限制:
   - 小电感: 同样电流下储能少 → 峰值电流更高
   - 需要更严格的 OCP (过流保护) 设置

3. 效率优化:
   - 不同电感的最优开关频率不同
   - 检测后可以微调控制环路参数

【检测原理】: 利用 IADPT 引脚的电阻分压
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

电路:
```

## 🔄 **特性七：PTM 直通模式 (Pass-Through Mode)**

### 📌 **什么是 PTM？**

```
PTM = Pass-Through Mode (直通模式)

TI 的专利技术!

【解决的问题】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

传统开关电源即使在轻负载也有损耗:
  - MOSFET 开关损耗 (虽然很小但存在)
  - 电感磁芯损耗 (磁滞 + 涡流)
  - 驱动电路损耗

当系统处于睡眠状态 (功耗只有 0.5W):
  这些固定损耗占比就很大了!
  
  例: 输入 5V/1A (5W), 系统只需 0.5W
  效率可能从 97% 降到 85%! (因为固定损耗不变)

PTM 的天才之处:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

正常模式 (Buck/Boost):
  VIN ──[Q1 开关]──┬── [L 电感] ──┬── [Q3 同步整流]── VOUT
                      │                │
                   开关损耗          开关损耗
                   电感损耗          驱动损耗

PTM 模式:
  VIN ────────────────┬─────────────────────── VOUT
                       │
                  [Q1 和 Q3 都常通!]  ← 就像一根导线!
                       │
                  几乎零损耗!!! (只有 MOSFET 导通电阻 I²R 损耗)
                       
【PTM 工作条件】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✅ 输入电压 ≈ 输出电压 (压差很小, < 0.5V)
   - 例如: 5V 输入, 系统需要 4.7V
   - 这种情况不需要降压/升压转换

✅ 负载很轻 (通常 < 10% of 最大负载)
   - 例如: 系统睡眠, 只需几百毫瓦

✅ 特殊寄存器配置 (见下方代码)

【PTM 的限制】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ 不能调节电压 (输入多少就是输出多少, 减去一点压降)
❌ 不能隔离噪声 (没有电感滤波, VBUS 纹波会直通到系统)
❌ 压差大时效率反而下降 (MOSFET 要承受大电流)

所以 PTM 只在特定条件下自动激活
```

### ⚙️ **PTM 配置方法**

```c
/**
 * @brief 配置 PTM (Pass-Through Mode)
 * @param enable: 1-启用 PTM, 0-禁用
 */
void configure_ptm_mode(uint8_t enable)
{
    if (!enable) {
        /* 禁用 PTM: 恢复正常的 Buck/Boost 工作 */
        BQ25710_ChargeOption1_t opt1;
        bq25710_read_reg16(BQ25710_REG_CHARGE_OPTION_1, &opt1.all);
        opt1.bits.EN_PTM = 0;  // 禁用 PTM
        bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_1, opt1.all);
        
        /* 将 ILIM_HIZ 引脚拉高 (退出 PTM 的必要条件之一) */
        // set_ilim_hiz_high();
        
        printf("[PTM] Pass-Through mode DISABLED\n");
        return;
    }
    
    printf("[PTM] Configuring Pass-Through mode...\n");
    
    /*
     * PTM 启用步骤 (严格按照 datasheet 9.4.3 节):
     */
    
    /* 步骤1: 禁用 EN_EXITILIM (ChargeOption2[7]) */
    BQ25710_ChargeOption2_t opt2;
    bq25710_read_reg16(BQ25710_REG_CHARGE_OPTION_2, &opt2.all);
    opt2.bits.PKPWR_RELAX_STAT = 0;  // 对应 EN_EXITILIM
    bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_2, opt2.all);
    
    /* 步骤2: 设置 RSNS_RSR = 1 (ChargeOption1[10]) */
    BQ25710_ChargeOption1_t opt1;
    bq25710_read_reg16(BQ25710_REG_CHARGE_OPTION_1, &opt1.all);
    opt1.bits.RSNS_RSR = 1;  // 20mΩ 检测电阻
    bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_1, opt1.all);
    
    /* 步骤3: 使能 PTM (ChargeOption1[2]) */
    opt1.bits.EN_PTM = 1;
    bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_1, opt1.all);
    
    /* 步骤4: 将 ILIM_HIZ 引脚接地 (硬件操作!) */
    /* 这是激活 PTM 的最后一步 */
    // set_ilim_hiz_low();  // GPIO 拉低
    
    printf("[PTM] Configuration complete.\n");
    printf("[PTM] PTM will activate when:\n");
    printf("[PTM]   - Input voltage ≈ System voltage (ΔV < 0.5V)\n");
    printf("[PTM]   - Load current is light (< 10%% of max)\n");
    printf("[PTM]   - ILIM_HIZ pin is held LOW\n");
    printf("[PTM]\n");
    printf("[PTM] To exit PTM manually:\n");
    printf("[PTM]   - Set EN_PTM = 0, OR\n");
    printf("[PTM]   - Pull ILIM_HIZ HIGH, OR\n");
    printf("[PTM]   - Trigger VINDPM condition\n");
}

/**
 * @brief 检查当前是否处于 PTM 模式
 */
uint8_t is_ptm_active(void)
{
    /* PTM 没有直接的状态位可以读, 但可以通过以下迹象推断: */
    
    BQ25710_ChargerStatus_t status;
    bq25710_get_charger_status(&status);
    
    /* 在 PTM 模式下:
     * - IN_FCHRG 和 IN_PCHRG 都应该是 0 (不在充电)
     * - IN_VINDPM 可能是 1 (特殊状态)
     * - AC_STAT = 1 (适配器存在)
     */
    
    if (status.bits.AC_STAT && 
        !status.bits.IN_FCHRG && 
        !status.bits.IN_PCHRG &&
        !status.bits.IN_OTG) {
        /* 很可能是 PTM 或空闲状态 */
        
        /* 进一步检查: 测量 VIN 和 VSYS 的差异 */
        uint16_t vbus, vsys;
        bq25710_adc_read_channel(BQ25710_ADC_VBUS, &vbus);
        bq25710_adc_read_channel(BQ25710_ADC_VSYS, &vsys);
        
        if (abs(vbus - vsys) < 500) {  // 压差小于 0.5V
            return 1;  // 高概率在 PTM 模式
        }
    }
    
    return 0;  // 不在 PTM 模式
}
```

## 🛡️ **特性八：两级电流限制 (Peak Power Mode)**

### 📌 **充分利用适配器的过载能力**

```
大多数适配器都能承受短时间的过载!

例: 一个标称 65W (20V/3.25A) 的适配器
  实际上可以承受:
  - 持续: 3.25A (65W) - 长时间安全
  - 短时:  5.0A  (100W) - 几十毫秒内没问题!
  - 瞬间:  7.5A  (150W) - 微秒级脉冲

Peak Power Mode 就是利用这个特性!
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

【工作机制】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

定义两个电流级别:
  ILIM1 = 正常持续电流限制 (来自 IIN_HOST 寄存器, 例 3.25A)
  ILIM2 = 过载电流限制 (ILIM1 的 110%~450%, 可编程)

工作时序:
  ┌────────────────────────────────────────────────────┐
  │                                                    │
  │  正常状态:                                        │
  │  IIN 保持在 ILIM1 以下                              │
  │  充电器正常工作, 电池也在充电                       │
  │                                                    │
  │  ↑↑↑ 负载突然增大 (CPU 涡轮!)                      │
  │                                                    │
  │  第一阶段: TOVLD 时间窗口                         │
  │  允许电流冲到 ILIM2 (例如 5A)                     │
  │  此时不充电 (暂停给电池)                            │
  │  所有功率都供给系统                                  │
  │  持续时间: TOVLD (可设 5ms/10ms/20ms/40ms)          │
  │                                                    │
  │  第二阶段: TMAX 时间窗口                           │
  │  电流回落到 ILIM1 (3.25A)                          │
  │  恢复给电池充电                                      │
  │  持续时间: TMAX - TOVLD                             │
  │                                                    │
  │  循环...直到负载恢复正常                             │
  │                                                    │
  └────────────────────────────────────────────────────┘

【时序图】:
  IIN (A)
    ^
 6 │     ┌──────┐
   │     │ILIM2 │ 5A 过载区
 5 │     │  (5A)│
   │     │      │
 4 │     │      │     ┌─────┐
   │     │      │     │ILIM1│ 3.25A 正常
 3 │─────┘      └─────┘     │
   │  ←TOVLD→│←────TMAX────→│
 2 │         │              │
   │         │              │
 1 └─────────┴──────────────┴────────────→ 时间
            ↑              ↑
         开始过载        恢复正常+充电
         
  ICHG (充电电流)
    ^
 0.5├───────────────────────┬────────────────────
    │                       │
    │   (暂停充电)          │ (恢复充电)
    │                       │
  0 └───────────────────────┴────────────────────
```

### ⚙️ **Peak Power Mode 配置**

```c
/**
 * @brief 配置两级电流限制 (峰值功率模式)
 * @param ilim2_percent: ILIM2 相对于 ILIM1 的百分比
 *   可选: 110, 130, 150, 170, 190, 210, 230, 250,
 *         270, 290, 310, 330, 350, 370, 390, 410,
 *         430, 450 (对应寄存器值 1~20, 步进 20%)
 * @param tovld_ms: 过载允许持续时间 (ms)
 *   可选: 5, 10, 20, 40
 * @param tmax_ms: 总周期时间 (ms)
 *   必须 > tovld_ms
 */
void configure_peak_power_mode(uint8_t ilim2_percent, 
                             uint8_t tovld_ms,
                             uint16_t tmax_ms)
{
    printf("[PKPWR] Configuring Peak Power Mode...\n");
    
    BQ25710_ChargeOption2_t opt2;
    bq25710_read_reg16(BQ25710_REG_CHARGE_OPTION_2, &opt2.all);
    
    /* 步骤1: 使能峰值功率模式 */
    opt2.bits.EN_PKPWR_VSYS = 1;  // VSYS 下冲触发 (可选)
    opt2.bits.EN_PKPWR_IDPM = 1;  // IIN 过冲触发 (推荐使能)
    
    /* 步骤2: 设置 ILIM2 百分比 */
    /* 寄存器值 = (percent - 110) / 20 + 1 */
    /* 例: percent=150 → value=(150-110)/20+1=3 */
    uint8_t reg_val = (ilim2_percent - 110) / 20 + 1;
    if (reg_val < 1) reg_val = 1;
    if (reg_val > 20) reg_val = 20;
    
    opt2.bits.PKPWR_TOVLD_DEG = reg_val;  // 复用字段存 ILIM2
    /* 注意: 实际 ILIM2 设置在 ProchotOption0[15:11] */
    
    bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_2, opt2.all);
    
    /* 步骤3: 设置 TOVLD (过载时间) */
    uint8_t tovld_code;
    switch (tovld_ms) {
        case 5:  tovld_code = 0b00; break;
        case 10: tovld_code = 0b01; break;
        case 20: tovld_code = 0b10; break;
        case 40: tovld_code = 0b11; break;
        default: tovld_code = 0b10; break;  // 默认 20ms
    }
    opt2.bits.PKPWR_TMAX = tovld_code;
    
    /* 步骤4: 设置 TMAX (总周期, 通过 TOVLD 间接实现) */
    /* 如果 TOVLD == TMAX, 则峰值模式始终开启 */
    /* 这里我们让 TMAX 比 TOVLD 大一些 */
    /* 实际 TMAX 由硬件根据 TOVLD 自动计算 */
    
    bq25710_write_reg16(BQ25710_REG_CHARGE_OPTION_2, opt2.all);
    
    /* 步骤5: 在 ProchotOption0 中设置实际的 ILIM2 阈值 */
    BQ25710_ProchotOption0_t popt0;
    bq25710_read_reg16(BQ25710_REG_PROCHOT_OPTION_0, &popt0.all);
    
    /* ILIM2_VTH: 范围 110%~450%, 步进 20%, 默认 150% */
    /* 寄存器值: 1=110%, 2=130%, ..., 17=430%, 18=450%, 19=250%, 20=270%... */
    popt0.bits.ILIM2_VTH = reg_val;
    
    bq25710_write_reg16(BQ25710_REG_PROCHOT_OPTION_0, popt0.all);
    
    printf("[PKPWR] Settings applied:\n");
    printf("[PKPWR]   ILIM1: Normal limit (from IIN_HOST register)\n");
    printf("[PKPWR]   ILIM2: %d%% of ILIM1 (overload threshold)\n", ilim2_percent);
    printf("[PKPWR]   TOVLD: %dms (overload duration)\n", tovld_ms);
    printf("[PKPWR]   TMAX: Auto-calculated based on TOVLD\n");
    printf("[PKPWR] \n");
    printf("[PKPWR] Example behavior (with IIN_HOST=3.25A):\n");
    float ilim1 = 3.25f;
    float ilim2 = ilim1 * ilim2_percent / 100.0f;
    printf("[PKPWR]   Normal operation: IIN ≤ %.2fA (charging enabled)\n", ilim1);
    printf("[PKPWR]   During burst:    IIN ≤ %.2fA (charging paused)\n", ilim2);
    printf("[PKPWR]   Burst duration:   %dms\n", tovld_ms);
}
```

## 🌡️ **特性九：热关断保护 (Thermal Shutdown)**

### 📌 **温度保护等级**

```
BQ25710 有多级温度保护:

Level 1: 警告 (无动作, 仅可监测)
  - 温度 > 100°C (可通过 NTC 读出)

Level 2: 降额 (自动降低性能)
  - 温度 > 120°C (内部逻辑, 用户不可配置)
  - 表现: 降低开关频率, 限制电流

Level 3: 热关断 (TSHUT - Thermal Shutdown)
  - 结温 > 155°C ⚠️
  - 动作: 完全停止转换器!
  
Level 4: 恢复
  - 结温降至 < 135°C
  - 动作: 软启动恢复 (逐步提升输出)

【热关断期间的系统行为】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

转换器关闭后:
  ├── LDO 电流限制降至 16mA (仅维持最低功能)
  ├── REGN LDO 关闭 (基准源消失)
  ├── CHRG_OK 拉低 (通知系统异常)
  ├── BATFET 可能保持导通 (如果电池能给系统供电)
  └── 所有 PWM 开关停止

恢复过程 (温度降到 135°C 以下):
  1. 等待 10ms 左右 (去抖动)
  2. 软启动: 逐步升高占空比 (避免冲击)
  3. 正常工作恢复
  4. 如果再次超温 → 再次关断 (循环直到根因消除)

【散热设计建议】:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. PCB 布局:
   - 芯片底部裸焊盘 (Exposed Pad) 必须充分焊接
   - 打 5~9 个过孔连到内地层 (帮助散热)
   - 周围不要有大面积铜皮阻挡空气流动

2. 外部散热:
   - 如果长时间 >3A 工作, 建议加散热片
   - 散热片材料: 铝或铜
   - 接触面使用导热硅脂

3. 空气对流:
   - 不要把芯片封闭在小盒子里
   - 保留通风孔
```

### ⚙️ **温度监测代码**

```c
/**
 * @brief 温度监测与保护 (基于 NTC 热敏电阻)
 */

/* NTC 10kΩ @25°C 的 Steinhart-Hart 方程系数 */
#define NTC_A  0.001308620f
#define NTC_B  0.000236268f
#define NTC_C  0.000000096788f

/**
 * @brief 将 ADC 的 CMPIN 电压转换为温度
 * @param cmpin_mv: CMPIN 引脚的 ADC 读数 (mV)
 * @return 温度值 (°C)
 */
float convert_ntc_to_temperature(uint16_t cmpin_mv)
{
    /* CMPIN 电压转电阻 */
    /* 电路: VDDA(2.3V或1.2V) -- [NTC] -- [固定电阻] -- GND */
    /* 假设使用 2.3V 基准, 10kΩ 下拉电阻 */
    
    float v_ref = 2300.0f;  // mV
    float r_fixed = 10000.0f;  // 10kΩ
    float r_ntc;
    
    /* 分压公式: V_cmpin = V_ref * R_fixed / (R_ntc + R_fixed) */
    /* 反推: R_ntc = R_fixed * (V_ref - V_cmpin) / V_cmpin */
    if (cmpin_mv <= 0 || cmpin_mv >= v_ref) {
        return -273.15f;  // 无效值返回绝对零度
    }
    
    r_ntc = r_fixed * (v_ref - cmpin_mv) / cmpin_mv;
    
    /* Steinhart-Hart 方程: 1/T = A + B*ln(R) + C*(ln(R))^3 */
    float log_r = log(r_ntc);
    float inv_temp = NTC_A + NTC_B * log_r + NTC_C * log_r * log_r * log_r;
    float temp_k = 1.0f / inv_temp;
    float temp_c = temp_k - 273.15f;
    
    return temp_c;
}

/**
 * @brief 执行温度保护策略
 */
void temperature_protection_task(void)
{
    uint16_t cmpin_raw;
    int ret = bq25710_adc_read_channel(BQ25710_ADC_CMPIN, &cmpin_raw);
    if (ret != BQ25710_OK) return;
    
    float temperature = convert_ntc_to_temperature(cmpin_raw);
    
    printf("[TEMP] Battery temperature: %.1f °C\n", temperature);
    
    /* 分级保护策略 */
    if (temperature < 0) {
        /* 低温: 严重警告 */
        printf("[TEMP] ⚠️ COLD WARNING: Below 0°C!\n");
        printf("[TEMP] Action: Reduce charge current to 128mA (trickle)\n");
        bq25710_set_charge_current(128);  // 极小电流预充
        
    } else if (temperature < 10) {
        /* 次低温: 降额充电 */
        printf("[TEMP] Low temperature zone (0-10°C)\n");
        printf("[TEMP] Action: Half charging current\n");
        bq25710_set_charge_current(1024);  // 1A
        
    } else if (temperature > 45) {
        /* 高温预警 */
        printf("[TEMP] ⚠️ HOT WARNING: Above 45°C!\n");
        printf("[TEMP] Action: Reduce current and monitor closely\n");
        bq25710_set_charge_current(1536);  // 1.5A
        
    } else if (temperature > 55) {
        /* 过热危险 */
        printf("[TEMP] 🔥 DANGER: Above 55°C!\n");
        printf("[TEMP] Action: STOP charging immediately!\n");
        bq25710_disable_charging();
        
        /* 可选: 强制进入低功耗减少自发热 */
        bq25710_set_low_power_mode(1);
        
    } else {
        /* 正常温度范围 (10-45°C): 全速充电 */
        static uint8_t last_current = 0;
        if (last_current != 2048) {
            bq25710_set_charge_current(2048);  // 恢复正常
            last_current = 2048;
        }
    }
    
    /* 检查芯片内部热关断状态 */
    BQ25710_ChargerStatus_t status;
    bq25710_get_charger_status(&status);
    
    static uint8_t thermal_shutdown_count = 0;
    
    /* 虽然无法直接读结温, 但可以通过反复出现的故障推断 */
    if (status.bits.FAULT_ACOC || status.bits.FAULT_BATOC) {
        /* 反复出现过流可能是热导致的 */
        thermal_shutdown_count++;
        
        if (thermal_shutdown_count > 5) {
            printf("[TEMP] 🔥 Possible THERMAL SHUTDOWN detected!\n");
            printf("[TEMP] Fault count: %d in short time\n", thermal_shutdown_count);
            printf("[TEMP] Recommendation: Check heatsink and airflow!\n");
            
            /* 重置计数器避免重复报警 */
            thermal_shutdown_count = 0;
        }
    }
}
```

```c
/**
 * @brief 从 BQ25710 获取最新数据供 GUI 显示
 */
void gui_data_fetcher(BatteryInfo_t *batt, ChargeParams_t *params)
{
    if (batt == NULL || params == NULL) return;
    
    BQ25710_ADC_Result_t adc;
    BQ25710_ChargerStatus_t status;
    
    /* 读取 ADC 数据 */
    if (bq25710_adc_read_all(&adc) == BQ25710_OK) {
        batt->voltage_mv = adc.vbat_mV;
        batt->current_ma = adc.ichg_mA;  /* 正数=充电 */
        
        /* 如果 ICHG ≈ 0 但 IDCHG > 0, 则是放电状态 */
        if (abs(batt->current_ma) < 50 && adc.idchg_mA > 0) {
            batt->current_ma = -adc.idchg_mA;  /* 负数表示放电 */
        }
        
        /* 计算电量百分比 (简化算法) */
        static const uint16_t full_voltage = 4200;  /* 1S满充电压 */
        static const uint16_t empty_voltage = 3000; /* 空电电压 */
        
        if (batt->voltage_mv >= full_voltage) {
            batt->percent = 100;
        } else if (batt->voltage_mv <= empty_voltage) {
            batt->percent = 0;
        } else {
            batt->percent = (uint8_t)((batt->voltage_mv - empty_voltage) * 100 /
                                           (full_voltage - empty_voltage));
        }
    }
    
    /* 读取充电器状态 */
    if (bq25710_get_charger_status(&status) == BQ25710_OK) {
        if (status.bits.IN_OTG) {
            batt->status = STATUS_DISCHARGING;
        } else if (status.bits.IN_FCHRG || status.bits.IN_PCHRG) {
            batt->status = STATUS_CHARGING;
        } else if (batt->percent >= 99) {
            batt->status = STATUS_CHARGE_COMPLETE;
        } else {
            /* 检查是否有适配器 */
            int adapter = bq25710_is_adapter_present();
            if (adapter > 0) {
                batt->status = STATUS_IDLE;  /* 有适配器但没在充 */
            } else {
                batt->status = STATUS_IDLE;  /* 无适配器 */
            }
        }
        
        /* 检查故障标志 */
        uint16_t fault_mask;
        bq25710_get_fault_status(&fault_mask);
        if (fault_mask != 0) {
            batt->status = STATUS_FAULT;
        }
    }
    
    /* 读取当前参数设置 */
    bq25710_get_charge_current(&params->target_current_ma);
    bq25710_get_max_charge_voltage(&params->target_voltage_mv);
    bq25710_get_input_current_limit(&params->input_limit_ma);
    bq25710_get_vindpm_threshold(&params->vindpm_threshold_mv);
    bq25710_get_min_sys_voltage_mv(&params->min_sys_voltage_mv);
    
    /* 模拟温度 (如果硬件支持 NTC) */
    batt->temperature_c = 35;  /* 示例值 */
    batt->health_percent = 95;  /* 示例值 */
}

/**
 * @brief 告警检查回调
 */
AlertInfo_t gui_alert_checker(void)
{
    AlertInfo_t alert = {0};
    
    /* 检查各种异常条件 */
    BQ25710_ChargerStatus_t status;
    if (bq25710_get_charger_status(&status) != BQ25710_OK) {
        return alert;  /* 无告警 */
    }
    
    uint16_t fault_mask;
    bq25710_get_fault_status(&fault_mask);
    
    if (fault_mask & (1 << 7)) {  /* ACOV */
        alert.is_alert = true;
        alert.alert_type = 1;
        strcpy(alert.message, "AC Overvoltage!");
        alert.alert_color = COLOR_DANGER;
        return alert;
    }
    
    if (fault_mask & (1 << 6)) {  /* ACOC */
        alert.is_alert = true;
        alert.alert_type = 2;
        strcpy(alert.message, "AC Overcurrent!");
        alert.alert_color = COLOR_WARNING;
        return alert;
    }
    
    if (fault_mask & (1 << 3)) {  /* SYS_SHORT */
        alert.is_alert = true;
        alert.alert_type = 3;
        strcpy(alert.message, "System Short!");
        alert.alert_color = COLOR_DANGER;
        return alert;
    }
    
    /* 电量过低警告 */
    if (s_batt_info.percent <= 10 && s_batt_info.status != STATUS_CHARGING) {
        alert.is_alert = true;
        alert.alert_type = 4;
        snprintf(alert.message, sizeof(alert.message), 
                 "Low Battery: %d%%", s_batt_info.percent);
        alert.alert_color = COLOR_BATTERY_LOW;
        return alert;
    }
    
    return alert;  /* 无告警 */
}
```
